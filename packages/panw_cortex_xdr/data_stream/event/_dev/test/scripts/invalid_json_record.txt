# A record field that is sent as a string but is not valid JSON, or that is
# valid JSON but not an object, must not send the document to the failure store.
# The failing field is dropped, the error is recorded, and the rest of the
# document, including other string-encoded record fields, is still parsed.
#
# This is a script test because the pipeline test runner treats any error.message
# as a failure and cannot expect one.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

install_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default data.json
! stderr .
cp stdout got.json

exec jq -s 'length' got.json
stdout '^3$'

# Both documents are marked as pipeline errors and keep the original event.
exec jq -s 'all(.event.kind == "pipeline_error")' got.json
stdout '^true$'
exec jq -s 'all(.tags | index("preserve_original_event") != null)' got.json
stdout '^true$'
exec jq -s 'all(.event.original | length > 0)' got.json
stdout '^true$'

# The error names the processor that failed.
exec jq -s -r '[.[0].error.message] | flatten | join(" ")' got.json
stdout 'with tag json_action_process_token_[0-9a-f]+ in pipeline'
exec jq -s -r '[.[1].error.message] | flatten | join(" ")' got.json
stdout 'with tag json_action_file_mode_[0-9a-f]+ in pipeline'

# Record 1: the invalid token is dropped, the valid fds string is parsed.
exec jq -s '.[0].panw_cortex.xdr.event.action.process | has("token")' got.json
stdout '^false$'
exec jq -s '.[0].panw_cortex.xdr.event.action.process.fds.stdin' got.json
stdout '^"0"$'

# Record 2: the invalid file mode is dropped, the valid token string is parsed.
exec jq -s '(.[1].panw_cortex.xdr.event.action.file // {}) | has("mode")' got.json
stdout '^false$'
exec jq -s '.[1].panw_cortex.xdr.event.action.process.token.user_id' got.json
stdout '^"S-1-5-18"$'

# Record 3: values that parse, or already are, but are not objects are dropped
# too, and each one is reported. The valid token string is still parsed.
exec jq -s '(.[2].panw_cortex.xdr.event.action.file // {}) | has("mode") or has("pe_info")' got.json
stdout '^false$'
exec jq -s '.[2].panw_cortex.xdr.event.action.process | has("fds")' got.json
stdout '^false$'
exec jq -s '.[2].panw_cortex.xdr.event.action.process.token.user_id' got.json
stdout '^"S-1-5-18"$'
exec jq -s -r '[.[2].error.message] | flatten | join(" ")' got.json
stdout 'Field json.action_file_mode is not an object'
stdout 'Field json.action_file_pe_info is not an object'
stdout 'Field json.action_process_fds is not an object'

# The rest of each document is still processed.
exec jq -s 'all(.panw_cortex.xdr.event.action.process.image_name == "cmd.exe")' got.json
stdout '^true$'

uninstall_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .
