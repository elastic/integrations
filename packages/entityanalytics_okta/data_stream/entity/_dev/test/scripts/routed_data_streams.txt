# Verify that entity-analytics documents are correctly routed from the entity
# data stream to the user and device data streams via routing rules.
#
# The mock Okta API provides:
#   - 1 user (Isaac Brock)                → entityanalytics_okta.user
#   - 2 devices (Example Device 1 and 2)  → entityanalytics_okta.device
#
# The minimal-state path does not produce marker events, so the entity data
# stream is empty after routing rules divert user and device documents.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock Okta API service via symlink to the package's deploy directory.
exec ln -s ${PACKAGE_ROOT}/_dev/deploy/docker entityanalytics_okta
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} entityanalytics_okta

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml ENTITY_DS

# Wait for user documents in the user data stream.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -timeout 5m logs-entityanalytics_okta.user-*
cp stdout user_docs.json
exec jq '[.hits.hits[]._source.user.id] | unique[]' user_docs.json
stdout '00ub0oNGTSWTBKOLGLNR'

# Verify device documents were routed to the device data stream.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -timeout 2m logs-entityanalytics_okta.device-*
cp stdout device_docs.json
exec jq '[.hits.hits[]._source.device.id] | .[0]' device_docs.json
! stdout 'null'
exec jq '[.hits.hits[]._source.asset.type] | unique | .[0]' device_docs.json
stdout 'okta_device'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${ENTITY_DS}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down entityanalytics_okta
