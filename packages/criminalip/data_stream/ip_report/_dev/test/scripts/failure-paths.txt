# Failure-path regressions (pipeline tests reject error.message).
[!external_stack] skip 'Requires a running Elastic stack.'
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default case-0.json
! stderr .
stdout 'Criminal IP request failed with HTTP status 403'
! stdout '&lt;|&gt;|&amp;'

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default case-1.json
! stderr .
stdout 'Criminal IP request or JSON decoding failed; check endpoint and connectivity\.'
! stdout '&lt;|&gt;|&amp;'

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default case-2.json
! stderr .
stdout '"original":'
stdout 'preserve_original_event'
stdout 'Earlier error <keep & preserve>'
! stdout '&lt;|&gt;|&amp;'

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default case-3.json
! stderr .
stdout '"original":'
! stdout '"spoofed"|"unexpected":|"authentication"'
stdout 'success'
! stdout '&lt;|&gt;|&amp;'

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default case-4.json
! stderr .
stdout '"original":'
stdout 'did not return a usable IP report'
! stdout '&lt;|&gt;|&amp;'

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default case-5.json
! stderr .
stdout '"original":'
stdout 'preserve_original_event'
! stdout '&lt;|&gt;|&amp;'

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default case-6.json
! stderr .
stdout '"original":'
stdout 'preserve_original_event'
! stdout '&lt;|&gt;|&amp;'

uninstall_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .
