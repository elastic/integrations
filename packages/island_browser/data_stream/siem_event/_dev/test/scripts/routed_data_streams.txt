# Verify SIEM v1 collection and routing from siem_event to audit and admin_actions.
#
# The mock SIEM API returns one batch with two NDJSON events (1 BrowserAudit,
# 1 AdminAction), then an empty batch. The CEL program GETs the batch, POSTs
# the pageId ack, and on the next in-cycle GET receives count=0.
#
# Expected: 1 document in island_browser.audit and 1 in island_browser.admin_actions.
# The siem_event collector stream should retain no routed documents.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} island-siem-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Wait for both routed destination streams to receive exactly one document each.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 15s -timeout 5m logs-island_browser.audit-*
cp stdout audit_docs.json
exec jq '[.hits.hits[]._source | select(.island_browser.audit != null)] | length' audit_docs.json
stdout '^1$'
exec jq -r '[.hits.hits[]._source.island_browser.audit.source] | unique[]' audit_docs.json
stdout 'BrowserAudit'
exec jq -r '[.hits.hits[]._source.event.id] | unique[]' audit_docs.json
stdout '0554e3ab-618a-4171-a5d7-33555ac4476b'

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 15s -timeout 5m logs-island_browser.admin_actions-*
cp stdout admin_docs.json
exec jq '[.hits.hits[]._source | select(.island_browser.admin_actions != null)] | length' admin_docs.json
stdout '^1$'
exec jq -r '[.hits.hits[]._source.island_browser.admin_actions.source] | unique[]' admin_docs.json
stdout 'AdminAction'
exec jq -r '[.hits.hits[]._source.event.id] | unique[]' admin_docs.json
stdout 'b2f2209f-756f-4065-8cfb-d6d1878bfa2e'

# Combined count across both destination data streams.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 15s -timeout 5m logs-island_browser.audit-*,logs-island_browser.admin_actions-*
cp stdout all_docs.json
exec jq '.hits.total.value' all_docs.json
stdout '^2$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down island-siem-mock
