# A failed Sign-on poll must not drop state.initial_interval.
#
# The first evaluation still has initial_interval from the input config. The
# program returns a new object on the HTTP error path and does not copy that
# key forward. The next evaluation has no cursor, so it reads
# state.initial_interval and fails with:
#   failed eval: ERROR: no such key: initial_interval

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} workday-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# First poll reaches the API and indexes the HTTP error.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -timeout 3m ${DATA_STREAM_NAME}

# Wait until the following interval evaluates the state returned by that poll.
sleep 20s

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -size 20 -want 2 -timeout 2m ${DATA_STREAM_NAME}
cp stdout got_docs.json

# The following evaluation must not fail because initial_interval was dropped.
exec jq -r '[.hits.hits[]._source.error.message // empty] | flatten | .[]' got_docs.json
! stdout 'no such key: initial_interval'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down workday-mock
