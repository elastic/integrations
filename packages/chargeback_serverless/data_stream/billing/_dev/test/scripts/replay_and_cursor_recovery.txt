# Verify CEL billing collector behavior across:
# 1. Failed-day HTTP 503 response: emits error event without advancing cursor past the failed day.
# 2. Retry & initial backfill: collects project charges plus residual unassociated charges.
# 3. Cursor restoration & 72-hour replay on a subsequent day rollover: updates reattributed
#    project charges and emits zero-valued replacement snapshots (total_ecu == 0.0) for both
#    cleared residuals and removed SKUs.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} billing-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Phase 1: First poll hits HTTP 503 (1 error doc), then retries the same day and emits:
# - proj-1 (es.search-vcu = 80.0)
# - deleted_or_unassociated_elasticsearch (es.search-vcu = 20.0)
# - deleted_or_unassociated_observability (oblt.ingest-vcu = 5.0)
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 4 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_phase1.json

exec jq -r '[.hits.hits[]._source.error.code // empty] | .[]' got_docs_phase1.json
stdout '^503$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "proj-1" and .sku == "es.search-vcu") | .total_ecu] | .[0]' got_docs_phase1.json
stdout '^80(\.0+)?$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_elasticsearch" and .sku == "es.search-vcu") | .total_ecu] | .[0]' got_docs_phase1.json
stdout '^20(\.0+)?$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_observability" and .sku == "oblt.ingest-vcu") | .total_ecu] | .[0]' got_docs_phase1.json
stdout '^5(\.0+)?$'

# Phase 2: Switch mock to replay payload where proj-1 is fully attributed (100.0 ECU),
# clearing the es.search-vcu residual to 0.0, and oblt.ingest-vcu is removed from /costs/items.
docker_down billing-mock
cp billing-mock/config-replay.yml billing-mock/config.yml
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} billing-mock

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config_replay.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 7 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_phase2.json

exec jq -r '[.hits.hits[]._source | select(.project_id == "proj-1" and .sku == "es.search-vcu") | .total_ecu] | max' got_docs_phase2.json
stdout '^100(\.0+)?$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_elasticsearch" and .sku == "es.search-vcu" and .total_ecu == 0)] | length' got_docs_phase2.json
stdout '^[1-9][0-9]*$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_observability" and .sku == "oblt.ingest-vcu" and .total_ecu == 0)] | length' got_docs_phase2.json
stdout '^[1-9][0-9]*$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down billing-mock
