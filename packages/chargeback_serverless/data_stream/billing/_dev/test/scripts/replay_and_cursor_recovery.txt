# Verify CEL billing collector behavior across:
# 1. Failed-day HTTP 400 response: emits error event without advancing cursor past the failed day.
# 2. Retry & initial single-day backfill: collects project charges plus residual unassociated charges and
#    persists recent_day_keys in the stream cursor.
# 3. Subsequent replay pass with preserved stream cursor: updates reattributed project charges and
#    emits zero-valued replacement snapshots (total_ecu == 0.0) for both cleared residuals and
#    SKUs removed from /costs/items, then gates further replays for the same UTC day.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'
[!exec:sh] skip 'Skipping test requiring absent sh command'

# Select a single ready closed UTC billing day regardless of wall-clock UTC hour:
# - If within 2 minutes of 00:00 UTC or 02:00 UTC, wait past the boundary so
#   initial_from and latest_ready_to do not shift mid-test.
# - Before 02:00 UTC, yesterday's window is gated by `req.to + 2h > now`, so
#   `lookbehind: 2` selects `[today - 2d, today - 1d)` (1 ready day).
# - From 02:00 UTC onward, `lookbehind: 1` selects `[today - 1d, today)` (1 ready day).
exec sh -c 'wait_sec=$(jq -nr "now | gmtime | if (.[3] == 1 or .[3] == 23) and .[4] >= 58 then ((60 - .[4]) * 60 - (.[5] | floor) + 1) else 0 end"); if [ "$wait_sec" -gt 0 ]; then sleep "$wait_sec"; fi; lb=$(jq -nr "now | gmtime | if .[3] < 2 then 2 else 1 end"); jq -Rr --arg lb "$lb" "gsub(\"__LOOKBEHIND__\"; \$lb)" test_config.tmpl.yaml > test_config.yaml'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} billing-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Across three 5s polls on the same policy (preserving cursor.recent_day_keys):
# - Poll 1: HTTP 400 on /costs/instances -> 1 error document (error.code == "400").
# - Poll 2 (initial backfill): 3 documents:
#     * proj-1 (es.search-vcu = 80.0)
#     * deleted_or_unassociated_elasticsearch (es.search-vcu = 20.0)
#     * deleted_or_unassociated_observability (oblt.ingest-vcu = 5.0)
# - Poll 3 (bounded replay pass with preserved recent_day_keys): 3 documents:
#     * proj-1 (es.search-vcu = 100.0)
#     * deleted_or_unassociated_elasticsearch (es.search-vcu = 0.0)
#     * deleted_or_unassociated_observability (oblt.ingest-vcu = 0.0)
# - Poll 4+: gated by replay_completed_to == latest_ready_to_str (0 additional documents).
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 7 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq -r '[.hits.hits[]._source.error.code // empty] | .[]' got_docs.json
stdout '^400$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "proj-1" and .sku == "es.search-vcu") | .total_ecu] | min' got_docs.json
stdout '^80(\.0+)?$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "proj-1" and .sku == "es.search-vcu") | .total_ecu] | max' got_docs.json
stdout '^100(\.0+)?$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_elasticsearch" and .sku == "es.search-vcu" and .total_ecu == 20)] | length' got_docs.json
stdout '^1$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_elasticsearch" and .sku == "es.search-vcu" and .total_ecu == 0)] | length' got_docs.json
stdout '^1$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_observability" and .sku == "oblt.ingest-vcu" and .total_ecu == 5)] | length' got_docs.json
stdout '^1$'

exec jq -r '[.hits.hits[]._source | select(.project_id == "deleted_or_unassociated_observability" and .sku == "oblt.ingest-vcu" and .total_ecu == 0)] | length' got_docs.json
stdout '^1$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down billing-mock
