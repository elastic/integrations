  BODY_FILE="$(mktemp)"
  HTTP_CODE="$(curl -sS -o "${BODY_FILE}" -w '%{http_code}' -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" "$@")" || {
    cat "${BODY_FILE}" >&2
    rm -f "${BODY_FILE}"
    return 22
  }
  if [ "${HTTP_CODE}" -ge 400 ]; then
    echo "HTTP ${HTTP_CODE} on $*" >&2
    cat "${BODY_FILE}" >&2
    rm -f "${BODY_FILE}"
    return 22
  fi
  cat "${BODY_FILE}"
  rm -f "${BODY_FILE}"
}

es_curl() {
  BODY_FILE="$(mktemp)"
  HTTP_CODE="$(curl -sS -o "${BODY_FILE}" -w '%{http_code}' -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" "$@")" || {
    cat "${BODY_FILE}" >&2
    rm -f "${BODY_FILE}"
    return 22
  }
  if [ "${HTTP_CODE}" -ge 400 ]; then
    echo "HTTP ${HTTP_CODE} on $*" >&2
    cat "${BODY_FILE}" >&2
    rm -f "${BODY_FILE}"
    return 22
  fi
  cat "${BODY_FILE}"
  rm -f "${BODY_FILE}"
}

ensure_checkpoint1() {
  T_PART="$1"
  TID="$(jq -r --arg part "$T_PART" '.transforms[] | select(.id | startswith("logs-chargeback_serverless." + $part + "-")) | .id' initial_transforms.json)"
  if [ -z "${TID}" ]; then
    echo "Transform not found for ${T_PART}" >&2
    return 1
  fi
  ci=0
  while [ "${ci}" -lt 45 ]; do
    ST_JSON="$(es_curl "${ES_HOST}/_transform/${TID}/_stats")"
    T_STATE="$(echo "${ST_JSON}" | jq -r '.transforms[0].state')"
    CP_NUM="$(echo "${ST_JSON}" | jq -r '.transforms[0].checkpointing.last.checkpoint // 0')"
    HAS_NEXT="$(echo "${ST_JSON}" | jq -r '.transforms[0].checkpointing.next != null')"
    if [ "${T_STATE}" = "failed" ]; then
      echo "${ST_JSON}" >&2
      return 1
    fi
    if [ "${CP_NUM}" -ge 1 ] && [ "${HAS_NEXT}" = "false" ]; then
      return 0
    fi
    if [ "${CP_NUM}" -eq 0 ] && [ "${HAS_NEXT}" = "false" ]; then
      curl -s -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}"         -X POST "${ES_HOST}/_transform/${TID}/_schedule_now" >/dev/null 2>&1 || true
    fi
    ci=$((ci + 1))
    sleep 1
  done
  es_curl "${ES_HOST}/_transform/${TID}/_stats" >&2
  return 1
}

run_batch_transform() {
  BT_ID="$1"
  BT_TMPL="$2"
  BT_SRC="$3"
  BT_DEST="$4"
  BT_PIPE="$5"
  BT_PAGE="$6"
  curl -s -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" -X POST "${ES_HOST}/_transform/${BT_ID}/_stop?force=true&wait_for_completion=true" >/dev/null 2>&1 || true
  curl -s -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" -X DELETE "${ES_HOST}/_transform/${BT_ID}" >/dev/null 2>&1 || true
  jq --arg tn "$BT_TMPL" --arg si "$BT_SRC" --arg di "$BT_DEST" --arg pl "$BT_PIPE" --argjson ps "$BT_PAGE" '
    (.transforms[] | select(.id | startswith("logs-chargeback_serverless." + $tn + "-"))) |
    {
      source: (.source + {index: ($si | split(","))}),
      pivot: .pivot,
      dest: {index: $di, pipeline: $pl},
      settings: ((.settings // {}) + {max_page_search_size: $ps})
    }
  ' all_transforms.json | es_curl -X PUT "${ES_HOST}/_transform/${BT_ID}" -H 'Content-Type: application/json' -d @- >/dev/null
  es_curl -X POST "${ES_HOST}/_transform/${BT_ID}/_start" >/dev/null
  bt_i=0
  while [ "${bt_i}" -lt 30 ]; do
    bt_state="$(es_curl "${ES_HOST}/_transform/${BT_ID}/_stats" | jq -r '.transforms[0].state')"
    if [ "${bt_state}" = "stopped" ]; then
      break
    fi
    if [ "${bt_state}" = "failed" ]; then
      es_curl "${ES_HOST}/_transform/${BT_ID}/_stats" >&2
      return 1
    fi
    bt_i=$((bt_i + 1))
    sleep 1
  done
  if [ "${bt_state}" != "stopped" ]; then
    es_curl "${ES_HOST}/_transform/${BT_ID}/_stats" >&2
    return 1
  fi
  es_curl -X POST "${ES_HOST}/${BT_DEST}/_refresh" >/dev/null
}

es_total=0
es_failed=0
for df in "${DATA_STREAM_ROOT}/../../kibana/dashboard"/*.json; do
  for q in $(jq -r '.attributes.panelsJSON[].embeddableConfig.esql_query[]? | @base64' "${df}"); do
    es_total=$((es_total + 1))
    raw_q="$(printf '%s' "${q}" | base64 -d)"
    req_json="$(jq -nc --arg q "${raw_q}" --arg ts "2020-01-01T00:00:00Z" --arg te "2030-01-01T00:00:00Z" '{query: $q, params: [{_tstart: $ts}, {_tend: $te}]}')"

    if [ "${code}" -ge 300 ]; then
      es_failed=$((es_failed + 1))
      echo "FAILED ES|QL in ${df}: $(cat esql_err.json)" >&2
    fi
  done
done
