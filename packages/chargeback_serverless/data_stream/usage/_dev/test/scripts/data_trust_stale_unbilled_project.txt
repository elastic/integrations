#!/bin/sh
set -eu

STACK_CFG="${CONFIG_PROFILES}/${PROFILE}/stack/config.json"
if [ -f "${STACK_CFG}" ]; then
  ES_HOST="$(jq -r '.elasticsearch_host' "${STACK_CFG}")"
  ES_USER="$(jq -r '.elasticsearch_username' "${STACK_CFG}")"
  ES_PASS="$(jq -r '.elasticsearch_password' "${STACK_CFG}")"
  CA_CERT="$(jq -r '.ca_cert_file' "${STACK_CFG}")"
else
  ES_HOST="https://localhost:9200"
  ES_USER="elastic"
  ES_PASS="changeme"
  CA_CERT="${CONFIG_PROFILES}/${PROFILE}/certs/ca-cert.pem"
fi

es_curl() {
  BODY_FILE="$(mktemp)"
  HTTP_CODE="$(curl -sS -o "${BODY_FILE}" -w '%{http_code}' -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" "$@")" || {
    cat "${BODY_FILE}" >&2
    rm -f "${BODY_FILE}"
    return 22
  }
  if [ "${HTTP_CODE}" -ge 400 ]; then
    echo "HTTP ${HTTP_CODE} on $*" >&2
    cat "${BODY_FILE}" >&2
    rm -f "${BODY_FILE}"
    return 22
  fi
  cat "${BODY_FILE}"
  rm -f "${BODY_FILE}"
}

cleanup() {
  curl -s -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" \
    -X DELETE "${ES_HOST}/_data_stream/metrics-chargeback_serverless.billing-e2e-health?ignore_unavailable=true" >/dev/null 2>&1 || true
  curl -s -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" \
    -X DELETE "${ES_HOST}/cb-e2e-src-conf,cb-e2e-src-credits,cb-e2e-src-bill,cb-e2e-src-ds,cb-e2e-src-overlap,cb-e2e-ds-stage1?ignore_unavailable=true" >/dev/null 2>&1 || true
  curl -s -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" \
    -X POST "${ES_HOST}/serverless_billing_cost_lookup-2,serverless_project_status_lookup-2,serverless_project_contribution_lookup-2/_delete_by_query?refresh=true" \
    -H 'Content-Type: application/json' \
    -d '{"query":{"terms":{"project_id":["proj-healthy","proj-stale-unbilled","proj-inactive","proj-legacy-only","proj-currency-flip","proj-grp-backfill","proj-grp-new"]}}}' >/dev/null 2>&1 || true
  curl -s -u "${ES_USER}:${ES_PASS}" --cacert "${CA_CERT}" \
    -X POST "${ES_HOST}/serverless_billing_credits_lookup-1/_delete_by_query?refresh=true" \
    -H 'Content-Type: application/json' \
    -d '{"query":{"term":{"ess.billing.organization_id":"org-health-test"}}}' >/dev/null 2>&1 || true
}
trap cleanup EXIT

    break

done
