# Verify CEL usage collector behavior across:
# 1. Failed-day discovery HTTP 400 response: emits error event without advancing cursor.last_day.
# 2. Retry & live usage collection: collects datastream and project_status events across
#    discovered Serverless projects.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} usage-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_usage_docs.json

exec jq -r '[.hits.hits[]._source.error.code // empty] | .[]' got_usage_docs.json
stdout '^400$'

exec jq -r '[.hits.hits[]._source | select(.doc_type == "datastream" and .project_id == "proj-1") | .datastream] | .[0]' got_usage_docs.json
stdout '^logs-app-default$'

exec jq -r '[.hits.hits[]._source | select(.doc_type == "project_status" and .project_id == "proj-1") | .last_status] | .[0]' got_usage_docs.json
stdout '^ok$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down usage-mock
