# Verify that an expired commercial JWT token triggers a fresh login request
# before collecting URLhaus data on the next polling interval.
#
# The mock login endpoint returns two responses in rolling sequence:
#   1. A short-lived token (expires: 1, always stale on the next interval)
#   2. A refreshed token (expires: 4102444800)
# The URLs endpoint returns a different URL on each request so we can verify
# both collection rounds succeeded after token refresh.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} abusech-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '([.hits.hits[]._source | select(.abusech.url != null)] | length) >= 2' got_docs.json
stdout '^true$'

exec jq '[.hits.hits[]._source | select(.error != null)] | length' got_docs.json
stdout '^0$'

exec jq '([.hits.hits[]._source.abusech.url.id] | unique | length) >= 2' got_docs.json
stdout '^true$'

exec jq -c '[.hits.hits[]._source.threat.indicator.provider] | unique | sort' got_docs.json
stdout '^\["after_refresh_reporter","before_refresh_reporter"\]$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down abusech-mock

-- test_config.yaml --
input: cel
vars:
  enable_request_tracer: false
  api_type: commercial
  username: test@example.com
  password: test-password
data_stream:
  vars:
    url: http://abusech-mock:8080
    interval: 30s
    initial_interval: 72h
    page_size: 100
    preserve_original_event: true
    ioc_expiration_duration: 5d
