# Install Armis 0.7.0, collect one alert, upgrade to the package under test,
# then collect a second alert from the same mock. The second document shows
# collection continues across the upgrade.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'
[!has_previous_release] skip 'No previous release to upgrade from.'
[breaking_change] skip 'Cannot upgrade across breaking change.'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} armis-mock

install_package_from_registry -profile ${CONFIG_PROFILES}/${PROFILE} ${PACKAGE_NAME} 0.7.0
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} -version 0.7.0 test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_pre_upgrade.json

exec jq -r '[.hits.hits[]._source.event.id] | .[]' got_docs_pre_upgrade.json
stdout '^71$'

add_package -profile ${CONFIG_PROFILES}/${PROFILE}
upgrade_package_latest -profile ${CONFIG_PROFILES}/${PROFILE}
stdout 'upgraded package '${PACKAGE_NAME}

docker_down armis-mock
cp armis-mock/config-phase2.yml armis-mock/config.yml
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} armis-mock

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_post_upgrade.json

exec jq -r '[.hits.hits[]._source.event.id] | sort | join(",")' got_docs_post_upgrade.json
stdout '^71,72$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down armis-mock
