# Device search returns 500 after a successful token response. The CEL
# program emits one error event and stops the poll before cursor update.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} armis-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 20s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq -r '[.hits.hits[]._source.error.code] | .[]' got_docs.json
stdout '^500$'

exec jq -r '[.hits.hits[]._source.error.message] | .[]' got_docs.json
stdout 'in:devices'

exec jq '[.hits.hits[]._source | select(.armis.device != null)] | length' got_docs.json
stdout '^0$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down armis-mock

-- test_config.yaml --
input: cel
vars:
  url: http://armis-mock:8090
  secret_key: test-secret-key
data_stream:
  vars:
    interval: 10m
    initial_interval: 24h
    page_size: 1
    http_client_timeout: 30s
    enable_request_tracer: false
-- armis-mock/docker-compose.yml --
version: '3.0'
services:
  armis-mock:
    image: docker.elastic.co/observability/stream:v0.18.0
    hostname: armis-mock
    ports:
      - 8090
    environment:
      PORT: "8090"
    volumes:
      - ./config.yml:/config.yml
    command:
      - http-server
      - --addr=:8090
      - --config=/config.yml
-- armis-mock/config.yml --
rules:
  - path: /api/v1/access_token/
    methods: ['POST']
    responses:
      - status_code: 200
        headers:
          Content-Type:
            - 'application/json'
        body: |-
          {"data":{"access_token":"token-a","expiration_utc":"2025-04-03T08:48:17.963452+00:00"},"success":true}
  - path: /api/v1/search/
    methods: ['GET']
    request_headers:
      Authorization:
        - "token-a"
    responses:
      - status_code: 500
        headers:
          Content-Type:
            - 'application/json'
        body: ''
