# Test that audit events on later pages are collected when paging is
# interrupted.
#
# Records are returned newest first. The mock returns the first page of the
# window, and fails the request for the second page once with a 401 (the input
# retries 5xx responses itself). The program set last_timestamp from the first
# page, so the next window started at the newest event and the second page was
# never requested in a window that holds it.
#
# A request for the second page in a window that starts at the first page's
# newest event gets an empty page, as from the API.
#
# This lives in a scripts test rather than a system test because the error
# marks the input degraded, which the system test runner treats as a failure.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} openai-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# The four events and one error document. Without the fix the count stays at 3:
# the first page and the error.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 5 -confirm 30s -timeout 3m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '[.hits.hits[]._source.event.id // empty] | unique | length' got_docs.json
stdout '^4$'
exec jq '[.hits.hits[]._source | select(.error.message != null)] | length' got_docs.json
stdout '^1$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down openai-mock

-- test_config.yaml --
input: cel
vars:
  url: http://openai-mock:8080
  admin_token: xxxx
data_stream:
  vars:
    batch_size: 2
    initial_interval: 1h
    interval: 10s
