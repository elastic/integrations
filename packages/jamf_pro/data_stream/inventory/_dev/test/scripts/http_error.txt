# The first request fails with a 401. Without the fix every later poll fails
# with "no such key" and nothing is collected.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} jamf-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# One error and two computers, steady over several polls.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 45s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '[.hits.hits[]._source | select(.error.message != null)] | length' got_docs.json
stdout '^1$'
exec jq -r '[.hits.hits[]._source.error.message // empty] | flatten | .[]' got_docs.json
stdout 'HTTP response code: 401'
exec jq '[.hits.hits[]._source | select(.jamf_pro.inventory != null)] | length' got_docs.json
stdout '^2$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down jamf-mock

-- test_config.yaml --
input: cel
vars:
  api_host: http://jamf-mock:8080
  client_id: test_client_id
  client_secret: test_client_secret
data_stream:
  vars:
    interval: 10s
    page_size: 2
    enable_request_tracer: false
    enable_section_hardware: false
    enable_section_operating_system: false
