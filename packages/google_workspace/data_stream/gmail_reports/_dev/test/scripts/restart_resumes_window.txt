# Test that a restart in the middle of a window resumes at the saved page
# instead of reading the whole window again.
#
# The Reports API returns the newest events first, so the timestamp cursor can
# only move after the last page of a window. Without the page token in the
# cursor, a restart (or a configuration change, which restarts the input) reads
# the open window again and duplicates its events.
#
# Phase 1: the mock serves page 1 (event 1) and page 2 (event 2). Page 2 points
# at a third page, which the mock answers with an empty page that carries the
# same token again, so the window never completes. An empty page publishes no
# event, so the test sees exactly two documents and the cursor published with
# page 2 holds the window bounds and the token for page 3.
#
# The agent container is then restarted. In phase 2 the mock serves page 3
# (event 3, the last page) only for that token. A request without a token, which
# is what a restart without the saved token would make, gets an empty answer.
#
# Expected outcome when the token is resumed: event ids 1, 2 and 3, each once.
# Expected failure without the fix: the window is read again after the restart
# so ids 1 and 2 appear twice, or id 3 never arrives.
#
# The service account key is for testing only. It is generated for each run,
# is only trusted by the mock server and protects nothing. No key is stored in
# the repository.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'
[!exec:openssl] skip 'Skipping test requiring absent openssl command'
[!exec:docker] skip 'Skipping test requiring absent docker command'

exec openssl genpkey -algorithm RSA -pkeyopt rsa_keygen_bits:2048 -out key.pem
exec jq -n --rawfile key key.pem --slurpfile creds credentials.json -f test_config.jq
cp stdout test_config.yaml

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -container_name AGENT_CONTAINER -network_name NETWORK_NAME

docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} gw-mock

add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Pages 1 and 2 are collected and the window then stalls on page 3.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 20s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_before_restart.json

exec jq -r '[.hits.hits[]._source.event.id | select(. != null)] | sort | join(",")' got_docs_before_restart.json
stdout '^1,2$'

# Restart the agent in the middle of the window. The container keeps its
# filesystem, so the input state directory survives. Allow a graceful stop so
# that the input's state is saved. Check that the restart took place, so that
# the test cannot pass by never restarting the agent.
exec docker inspect -f '{{.State.StartedAt}}' ${AGENT_CONTAINER}
cp stdout started_before.txt
exec docker restart -t 60 ${AGENT_CONTAINER}
exec docker inspect -f '{{.State.StartedAt}}' ${AGENT_CONTAINER}
cp stdout started_after.txt
! cmp started_before.txt started_after.txt

# Let the last page be served from now on.
docker_down gw-mock
cp gw-mock/config-phase2.yml gw-mock/config.yml
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} gw-mock

# The agent needs time to start again, and then polls every 10s. In a local run
# it made its first request about 3s after the restart, so 60s leaves a wide
# margin for a slower machine.
sleep 60s

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 0s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_after_restart.json

# Each event once, and no event twice.
exec jq -r '[.hits.hits[]._source.event.id | select(. != null)] | sort | join(",")' got_docs_after_restart.json
stdout '^1,2,3$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down gw-mock
