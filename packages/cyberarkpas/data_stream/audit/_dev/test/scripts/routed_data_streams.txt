# Verify that VaultMonitor messages are routed from the audit data stream to
# the monitor data stream via routing rules.
#
# The embedded TCP sender provides:
#   - 1 VaultMonitor record  → cyberarkpas.monitor
#   - 1 Vault/Logon record   → cyberarkpas.audit (retained)

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the TCP sender with sample logs.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} cyberarkpas-tcp

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy with TCP input.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml AUDIT_DS

# Signal the sender to start streaming.
docker_signal cyberarkpas-tcp SIGHUP

# Verify the Vault/Logon record stays in the audit data stream.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 5s -timeout 5m ${AUDIT_DS}
cp stdout audit_docs.json
exec jq -r '.hits.hits[0]._source.observer.product' audit_docs.json
stdout '^Vault$'

# Verify the VaultMonitor record was routed to the monitor data stream.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -timeout 2m logs-cyberarkpas.monitor-*
cp stdout monitor_docs.json
# Filter by the current run's namespace (derived from AUDIT_DS) to avoid
# matching stale docs left by a previous run in the wildcard data stream.
exec jq -r --arg src ${AUDIT_DS} '($src | ltrimstr("logs-cyberarkpas.audit-")) as $ns | [.hits.hits[] | select(._source.data_stream.namespace == $ns) | ._source.observer.product][0]' monitor_docs.json
stdout '^VaultMonitor$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${AUDIT_DS}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down cyberarkpas-tcp
