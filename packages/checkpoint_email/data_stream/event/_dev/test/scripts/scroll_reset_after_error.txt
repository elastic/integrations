# Test that an error during pagination clears the scroll ID.
#
# If an error path keeps scroll_id in state while setting want_more to
# false, the next cycle builds a new date window but still sends the old
# scrollId. If that request fails, every later cycle sends the same stale
# scroll and fails the same way.
#
# Mock design:
#   Auth endpoint: always token-scroll.
#   Event query with scrollId scroll-one: always 400 (not retried by the
#   HTTP client).
#   Event query without scrollId: cycles [page with event D1 and
#   scrollId scroll-one, page with event D2 and no scrollId].
#
# With the fix:
#   1. query            -> event D1, scrollId scroll-one
#      query(scroll-one) -> 400 -> error event, scroll_id cleared
#   2. query            -> event D2
#
# Without the fix:
#   1. same as above, but scroll_id is kept
#   2. query(scroll-one) -> 400
#   ... repeats, event D2 never appears.
#
# Assertion: event D2 exists.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} checkpoint-email-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Wait for 3 documents: event D1 + 1 error + event D2.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

# Verify the failed scroll request was reported.
exec jq -r '[.hits.hits[]._source.error.message // empty] | flatten | .[]' got_docs.json
stdout '^POST /app/hec-api/v1.0/event/query: .*Invalid scroll'

# Verify event D2 was collected by a new query after the error.
exec jq '[.hits.hits[]._source | select(.event.id == "dbaabcdef01234567894115b9e64da71")] | length' got_docs.json
stdout '^1$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down checkpoint-email-mock
