# Test that a failed authentication request does not break later cycles.
#
# The CEL input keeps the state returned by the program between periodic
# runs. If the auth failure path returns a bare error object instead of
# merging it into state, client_id, client_secret and initial_interval are
# lost, and every later cycle fails with "no such key: client_id".
#
# Mock design:
#   Auth endpoint: cycles [401, token-recovered]. A 5xx would be retried by
#   the HTTP client and never reach the program, so a 401 is used.
#   Event query with token-recovered: always 200 with two events.
#
# With the fix (2 cycles):
#   1. auth(401)             -> error event, config fields kept in state
#   2. auth(token-recovered) -> 200 with two events -> events indexed
#
# Without the fix:
#   1. auth(401)             -> error event, config fields dropped
#   2. failed eval: no such key: client_id
#   ... repeats, the two events never appear.
#
# Assertion: both events exist and no failed evaluation was reported.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} checkpoint-email-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Wait for 3 documents: 1 auth error + 2 events.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

# Verify the auth failure was reported and names the auth endpoint.
exec jq -r '[.hits.hits[]._source.error.message // empty] | flatten | .[]' got_docs.json
stdout '^POST /auth/external: .*Invalid client credentials'

# Verify no cycle failed because the configuration was lost from state.
! stdout 'no such key'

# Verify both events were collected after re-authentication.
exec jq '[.hits.hits[]._source | select(.event.id == "c6d8674a04c30123456789e4d3ebd98" or .event.id == "cbaabcdef01234567894115b9e64da71")] | length' got_docs.json
stdout '^2$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down checkpoint-email-mock

    preserve_original_event: true
