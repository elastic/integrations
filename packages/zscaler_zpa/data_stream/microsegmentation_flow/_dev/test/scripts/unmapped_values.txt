# Keys the pipeline doesn't map are kept under zscaler_zpa.microsegmentation_flow.*,
# with their names converted to snake_case. These fields aren't
# defined in the field files, so the pipeline test runner would reject them.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

install_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .

# The input has the agent's host.name, as it would without the forwarded tag.
exec jq -cR '{message: ., tags: ["forwarded"], host: {name: "elastic-agent-host"}, data_stream: {type: "logs", dataset: "zscaler_zpa.microsegmentation_flow", namespace: "ep"}, _conf: {strict_fields: true}}' records.ndjson
cp stdout data.json

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default data.json
! stderr .
cp stdout got.json

exec jq -s 'length' got.json
stdout '^1$'

# Records from the Elastic template are only checked by sourcetype and
# version, so the extra key isn't an error.
exec jq -s 'map(has("error")) | any' got.json
stdout '^false$'

# An unrecognised Direction leaves network.direction unset and is kept in the
# defined direction field; an unknown key is kept, under its snake_case name.
exec jq -s -r '.[0] | [(.network | has("direction")), .zscaler_zpa.microsegmentation_flow.direction, .zscaler_zpa.microsegmentation_flow.event_type] | map(tostring) | join(",")' got.json
stdout '^false,LATERAL,flow_summary$'

# The rest of the record is processed, and ResourceName replaces the agent's
# host.name.
exec jq -s -r '.[0] | [.event.kind, .source.ip, .destination.ip, .network.transport, .event.action, .event.reason, .host.name] | join(",")' got.json
stdout '^event,192\.0\.2\.10,192\.0\.2\.20,tcp,allow,RULE,srv-app-01\.example\.com$'

exec jq -s 'map(has("_conf") or has("json")) | any' got.json
stdout '^false$'

# Keys with the same snake_case name: the first in sorted order is kept, and the
# other is dropped and reported.
exec jq -cR '{message: ., tags: ["forwarded"], data_stream: {type: "logs", dataset: "zscaler_zpa.microsegmentation_flow", namespace: "ep"}, _conf: {strict_fields: true}}' collision.ndjson
cp stdout collision.json

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default collision.json
! stderr .
cp stdout got_collision.json

exec jq -s -r '.[0].error.message | length, .[]' got_collision.json
stdout '^1\ndropped key .foo_bar. because its snake_case name .foo_bar. is already used by key .FooBar.$'
exec jq -s -r '.[0] | [.zscaler_zpa.microsegmentation_flow.foo_bar, .source.ip, (.tags | index("preserve_original_event") != null)] | map(tostring) | join(",")' got_collision.json
stdout '^upper,192\.0\.2\.10,true$'

uninstall_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .
