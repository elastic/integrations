# Values that cannot be converted are dropped and reported in error.message by
# the processor that failed, while the rest of the record is still processed.
# The pipeline test runner fails any document with error.message, so these
# cases cannot be pipeline tests.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

install_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .

# 0: invalid source IP, a non-numeric entry in the source port list and a
#    non-numeric destination port.
# 1: unparseable log and connection start timestamps, and an invalid
#    destination IP.
exec jq -cR '{message: ., tags: ["forwarded"], data_stream: {type: "logs", dataset: "zscaler_zpa.microsegmentation_flow", namespace: "ep"}, _conf: {strict_fields: true}}' records.ndjson
cp stdout data.json

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default data.json
! stderr .
cp stdout got.json

exec jq -s 'length' got.json
stdout '^2$'

# These are processing errors, not pipeline failures.
exec jq -s -r '.[].event.kind' got.json
stdout -count=2 '^event$'

exec jq -s -r '.[0].error.message | length, .[]' got.json
stdout '^3\nProcessor convert with tag convert_json_source_ip_to_source_ip failed with message: .not-an-ip. is not an IP string literal\.\ncould not parse source_ports value .abc. as a port number\nProcessor convert with tag convert_json_destination_port_to_destination_port failed with message: For input string: "https"$'

# The valid source port is kept, but is not used as source.port because the
# list had more than one entry.
exec jq -s -r '.[0] | [(.source | has("ip")), (.source | has("port")), (.destination | has("port")), (.zscaler_zpa.microsegmentation_flow.source_ports | map(tostring) | join(" "))] | map(tostring) | join(",")' got.json
stdout '^false,false,false,45566$'

# The rest of the record is processed.
exec jq -s -r '.[0] | [.["@timestamp"], .event.start, .destination.ip, (.related.ip | join(" ")), .network.transport, .event.action, .event.outcome, .host.name, .rule.name] | join(",")' got.json
stdout '^2026-11-06T09:00:00\.000Z,2026-11-02T15:40:13\.463Z,192\.0\.2\.20,192\.0\.2\.20,tcp,allow,success,srv-app-01\.example\.com,example_app_to_api_allow$'

exec jq -s -r '.[1].error.message | length, .[]' got.json
stdout '^3\nProcessor date with tag date_json_log_timestamp_to_timestamp failed with message: Text .not a date. could not be parsed at index 0\nProcessor date with tag date_json_connection_start_time_to_event_start failed with message: Text .yesterday. could not be parsed at index 0\nProcessor convert with tag convert_json_destination_ip_to_destination_ip failed with message: .192\.0\.2\.999. is not an IP string literal\.$'

exec jq -s -r '.[1] | [has("@timestamp"), (.event | has("start")), (.destination | has("ip")), .source.ip, .destination.port, (.zscaler_zpa.microsegmentation_flow.source_ports | length), .host.name] | map(tostring) | join(",")' got.json
stdout '^false,false,false,192\.0\.2\.10,1443,5,srv-app-01\.example\.com$'

# Records with errors keep the original event.
exec jq -s 'map((.tags | index("preserve_original_event") != null) and (.event.original | type == "string")) | all' got.json
stdout '^true$'

exec jq -s 'map(has("_conf") or has("json")) | any' got.json
stdout '^false$'

uninstall_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .
