# A message that is not valid JSON, or is JSON but not an object, must be
# indexed as a pipeline_error document that keeps the raw payload, so the
# failure can be diagnosed from the document itself.
# The pipeline test runner fails any document with error.message, so these
# cases cannot be pipeline tests.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

install_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .

# 0: truncated record.
# 1: record with a comma where a colon should be.
# 2: JSON array.
# 3: JSON string.
exec jq -cR '{message: ., tags: ["forwarded"], data_stream: {type: "logs", dataset: "zscaler_zpa.microsegmentation_flow", namespace: "ep"}, _conf: {strict_fields: true}}' records.ndjson
cp stdout data.json

simulate -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT} default data.json
! stderr .
cp stdout got.json

exec jq -s 'length' got.json
stdout '^4$'

# All reach the pipeline-level on_failure handler.
exec jq -s -r '.[].event.kind' got.json
stdout -count=4 '^pipeline_error$'
exec jq -s 'map(.error.message | length == 1) | all' got.json
stdout '^true$'

# Invalid JSON fails in the decode step; JSON that is not an object reaches the
# guard that requires an object.
exec jq -s -r '.[0:2][] | .error.message[0]' got.json
stdout -count=2 '^Processor .json. with tag .json_event_original_to_json. in pipeline .* failed with message '
exec jq -s -r '.[0].error.message[0]' got.json
stdout 'Unexpected end-of-input'
exec jq -s -r '.[1].error.message[0]' got.json
stdout 'was expecting a colon'
exec jq -s -r '.[2:4][] | .error.message[0]' got.json
stdout -count=2 '^Processor .conditional. with tag .fail_json_not_object. in pipeline .* failed with message .expected event\.original to be a JSON object.$'

# The raw payload is kept and tagged for preservation.
exec jq -s --rawfile raw records.ndjson '[.[].event.original] == ($raw | split("\n") | map(select(. != "")))' got.json
stdout '^true$'
exec jq -s 'map(.tags | index("preserve_original_event") != null) | all' got.json
stdout '^true$'

# Nothing is parsed, and the internal fields are removed.
exec jq -s -c 'map(keys) | unique' got.json
stdout '^\[\["data_stream","ecs","error","event","tags"\]\]$'
exec jq -s -c 'map(.event | keys) | unique' got.json
stdout '^\[\["kind","original"\]\]$'

uninstall_pipelines -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_ROOT}
! stderr .
