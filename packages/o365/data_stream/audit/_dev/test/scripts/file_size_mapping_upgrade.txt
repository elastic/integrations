# Test that upgrading from a release that dynamically mapped file.size as a
# keyword gives new documents a long mapping, and that nothing else changes
# type across the upgrade.
#
# Release 3.8.2 started passing numbers from the API through to the pipeline as
# strings, and dropped the conversion of file.size to a long. file.size had no
# explicit mapping, so on every backing index created after that upgrade it was
# dynamically mapped as a keyword, which conflicts with the long mapping on
# earlier indices and breaks rules and visualizations. The fix declares the
# field explicitly. Because a keyword cannot be changed to a long in place,
# Fleet must roll the data stream over when it applies the new mappings, so the
# post-upgrade documents must land in a new backing index.
#
# The pre-upgrade version is pinned to 3.11.1, the last release with the bug,
# rather than derived from LATEST_EPR_VERSION. Once the fix is published the
# latest release no longer exhibits the problem and an upgrade from it would
# prove nothing.
#
# The generic type comparison at the end is deliberately broad: it lists every
# field whose indexed type differs between the pre- and post-upgrade documents.
# The only permitted difference is file.size.
#
# See https://github.com/elastic/integrations/issues/21137.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'
[!env:LATEST_EPR_VERSION] skip 'no EPR release to upgrade from'
[!is_latest_version] skip 'dev version is not newer than latest EPR release'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock O365 API server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} o365-mock

# Install the last affected release from the registry and create a policy on it.
install_package_from_registry -profile ${CONFIG_PROFILES}/${PROFILE} ${PACKAGE_NAME} 3.11.1
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} -version 3.11.1 test_config.yaml DATA_STREAM_NAME

# One event with FileSizeBytes in scientific notation.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout before.json

# Document the bug being upgraded away from: the affected release indexes the
# raw string into a dynamically mapped keyword, so the indexed value is still
# in scientific notation.
exec jq -r '.hits.hits[0].fields["file.size"][0]' before.json
stdout '^1\.3405e4$'
exec jq -r '.hits.hits[0]._index' before.json
stdout '-000001$'

# Install the dev version from disk and upgrade the policy to it.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
upgrade_package_latest -profile ${CONFIG_PROFILES}/${PROFILE}
stdout 'upgraded package '${PACKAGE_NAME}

# Serve new events from the upgraded integration.
docker_down o365-mock
cp o365-mock/config-v2.yml o365-mock/config.yml
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} o365-mock

# One pre-upgrade document plus two new ones.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout after.json

# No pipeline failures and nothing dropped by ignore_malformed.
exec jq '[.hits.hits[]._source | select(.error != null)] | length' after.json
stdout '^0$'
exec jq '.aggregations.all_ignored.doc_count' after.json
stdout '^0$'

# The new documents are indexed as long with the expected integer values.
exec jq -r '[.hits.hits[] | select(.fields["event.id"][0] | startswith("sp-new")) | "\(.fields["event.id"][0])=\(.fields["file.size"][0])"] | sort | join(",")' after.json
stdout '^sp-new-enotation=13405,sp-new-plain=2048$'

# Fleet rolled the data stream over: the new documents are in the second
# generation backing index, not the one that holds the keyword mapping.
exec jq -r '[.hits.hits[] | select(.fields["event.id"][0] | startswith("sp-new")) | ._index] | unique | .[]' after.json
stdout '-000002$'
! stdout '-000001$'

# The pre-upgrade document is untouched and still a keyword in its own index.
exec jq -r '.hits.hits[] | select(.fields["event.id"][0] == "sp-old") | "\(.fields["file.size"][0]) \(._index)"' after.json
stdout '^1\.3405e4 .*-000001$'

# Generic check: for every field present both before and after the upgrade,
# compare the JSON type of the indexed value. The only field allowed to change
# type is file.size.
exec jq -S '[.hits.hits[].fields | to_entries[] | {key, value: (.value[0] | type)}] | from_entries' before.json
cp stdout types_before.json
exec jq -S '[.hits.hits[] | select(.fields["event.id"][0] | startswith("sp-new")) | .fields | to_entries[] | {key, value: (.value[0] | type)}] | from_entries' after.json
cp stdout types_after.json
exec jq -c -n --slurpfile a types_before.json --slurpfile b types_after.json '[$a[0] | to_entries[] | select($b[0][.key] != null and $b[0][.key] != .value) | .key]'
stdout '^\["file.size"\]$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down o365-mock
