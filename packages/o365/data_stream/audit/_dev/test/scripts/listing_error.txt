# Test that a listing request the API rejects is reported once and skipped.
#
# The mock answers every content listing request with a 400, as the API does
# for a time range it will not serve. (The input retries 5xx responses itself.) The listing error
# branch said it would "skip this link and continue", but it returned its error
# as a single object, as the content fetch error branch did before it was
# changed to a list. The runtime discards the cursor returned with a single
# object, so the link stayed at the head of the queue, and since the branch
# sets want_more, the same listing was requested again on every evaluation, up
# to max_executions times per interval.
#
# This lives in a scripts test rather than a system test because the error
# document carries event.kind pipeline_error, which the system test runner
# treats as a failure.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} o365-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# One error document for the rejected listing. Without the fix the listing is
# requested again within the same interval until max_executions, so the count
# climbs past 1.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 30s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq -r '[.hits.hits[]._source.error.message // empty] | flatten | .[]' got_docs.json
stdout 'activity/feed/subscriptions/content'

# Exactly one error document, not one per evaluation.
exec jq '[.hits.hits[]._source | select(.error.message != null)] | length' got_docs.json
stdout '^1$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down o365-mock

-- test_config.yaml --
input: cel
vars: ~
data_stream:
  vars:
    url: http://o365-mock:8080
    token_url: http://o365-mock:8080
    preserve_original_event: true
    client_id: test-cel-client-id
    client_secret: test-cel-client-secret
    azure_tenant_id: test-cel-tenant-id
    content_types: "Audit.Exchange"
    # One poll only, so that a second poll's listing does not add an error
    # document for a reason unrelated to the behaviour under test.
    interval: 10m
    # Bound a regression: without the fix the rejected listing is requested
    # again within the same interval until this limit.
    max_executions: 20
    initial_interval: 1h
    enable_request_tracer: false

  - path: /api/v1.0/test-cel-tenant-id/activity/feed/subscriptions/content

        - "Bearer test-token"
