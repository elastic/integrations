# Test that file size values are indexed with their declared types for every
# shape the Office 365 Management Activity API is known to send.
#
# The API returns byte counts as JSON numbers, sometimes in scientific notation
# (1.3405e4) or with a fractional part (1000.0), and occasionally as JSON
# strings. The CEL program decodes numbers as strings, so the ingest pipeline
# receives e.g. "1.3405e4" and must make sure that what reaches Elasticsearch
# is indexed as a long, not dynamically mapped as a keyword.
#
# This cannot be a pipeline test: pipeline tests only see the _source emitted by
# the pipeline, and elastic-package accepts a numeric string for a long field,
# so a "1000.0" in file.size passes. It cannot be a system test either, because
# system tests validate against the package's field definitions, which treat
# every ECS field as defined whether or not Elasticsearch has a mapping for it.
# Only the values returned by the search `fields` API reflect the mapping that
# Elasticsearch actually applied, which is what this test asserts on.
#
# See https://github.com/elastic/integrations/issues/21137.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock O365 API server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} o365-mock

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Seven SharePoint events exercising the size field shapes, plus one Exchange
# DLP event carrying sizes under ExchangeMetaData.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 8 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

# No pipeline failures.
exec jq '[.hits.hits[]._source | select(.error != null)] | length' got_docs.json
stdout '^0$'

# No field on any document was dropped by ignore_malformed. Fleet logs data
# streams set index.mapping.ignore_malformed: true, so a value that cannot be
# parsed into its mapped type is silently skipped and recorded in _ignored
# rather than rejecting the document.
exec jq '.aggregations.all_ignored.doc_count' got_docs.json
stdout '^0$'

# file.size is ECS long. Every value, whatever its input shape, must be indexed
# as a number with the expected integer value. A keyword mapping would leave
# "1.3405e4" and "1000.0" as strings here.
exec jq -r '[.hits.hits[] | select(.fields["file.size"] != null) | "\(.fields["event.id"][0])=\(.fields["file.size"][0])"] | sort | join(",")' got_docs.json
stdout '^sp-both=4096,sp-decimal=1000,sp-enotation=13405,sp-filesize-enotation=1000000000,sp-filesize-string=35920,sp-plain=2048$'

exec jq -c '[.hits.hits[].fields["file.size"][0] // empty | type] | unique' got_docs.json
stdout '^\["number"\]$'

# The event with no size fields must not get a file.size.
exec jq '[.hits.hits[] | select(.fields["event.id"][0] == "sp-none") | .fields["file.size"]] | map(. != null) | any' got_docs.json
stdout '^false$'

# o365.audit.FileSizeBytes is declared long in fields.yml; Elasticsearch coerces
# the scientific-notation string on indexing.
exec jq -c '[.hits.hits[] | select(.fields["event.id"][0] == "sp-enotation") | .fields["o365.audit.FileSizeBytes"][0]]' got_docs.json
stdout '^\[13405\]$'

# o365.audit.FileSize is declared keyword. The normalization script must have
# rewritten the scientific notation to a plain integer string.
exec jq -r '.hits.hits[] | select(.fields["event.id"][0] == "sp-filesize-enotation") | .fields["o365.audit.FileSize"][0]' got_docs.json
stdout '^1000000000$'

# o365.audit.ExchangeMetaData.* is declared long.
exec jq -c '[.hits.hits[] | select(.fields["event.id"][0] == "ex-dlp") | .fields["o365.audit.ExchangeMetaData.FileSize"][0], .fields["o365.audit.ExchangeMetaData.RecipientCount"][0]]' got_docs.json
stdout '^\[13405,2\]$'

# Generic checks over every field of every document.
#
# No field is indexed as a string while holding a number in _source. This is
# the signature of a numeric field that has no explicit mapping and was
# dynamically mapped as keyword.
exec jq -c '[.hits.hits[] | ._source as $s | .fields | to_entries[] | select((.value[0] | type) == "string") | select(((try ($s | getpath(.key | split("."))) catch null) | type) == "number") | .key] | unique' got_docs.json
stdout '^\[\]$'

# No keyword field holds a number in scientific notation. Fields that are
# numbers at the API but keywords in the package must be normalized so that
# searching for the value works.
exec jq -c '[.hits.hits[].fields | to_entries[] | select((.value[0] | type) == "string") | select(.value[0] | test("^-?[0-9]*\\.?[0-9]+[eE][-+]?[0-9]+$")) | .key] | unique' got_docs.json
stdout '^\[\]$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down o365-mock
