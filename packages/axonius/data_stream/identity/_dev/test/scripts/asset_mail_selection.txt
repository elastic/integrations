# One user asset with four adapter records: two case variants of one
# address, a different address, and one record with no mail. Every
# record must get the same user.email: the lexical minimum of the
# lowercased addresses. A case-sensitive minimum would pick the
# all-caps variant instead.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} axonius-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 4 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

# All four adapter records carry user.email.
exec jq '[.hits.hits[]._source | select(.user.email != null)] | length' got_docs.json
stdout '^4$'

# And it is the same lowercased address on every record.
exec jq -c '[.hits.hits[]._source.user.email] | unique' got_docs.json
stdout '^\["alice\.smith@example\.com"\]$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down axonius-mock
