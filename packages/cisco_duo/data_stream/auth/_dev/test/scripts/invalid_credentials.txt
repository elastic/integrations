# Test that an invalid/insufficient API key (non-200 response from the Cisco
# Duo Admin API) is handled by the CEL collector-error path and the ingest
# pipeline's terminate_collector_error branch: the error event is indexed with
# its error.* fields preserved, no auth data is collected, and no
# event.kind: pipeline_error is produced (regression test for issue #22383).
#
# The Cisco Duo API returns a non-200 with a JSON error body when the
# integration/secret key is wrong. The auth CEL program maps that response
# to a single error event with want_more=false, which the ingest pipeline must
# terminate before attempting to parse it as a log.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock Cisco Duo Admin API server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} duo-mock

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy pointing at the mock server with wrong credentials.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Wait for exactly 1 document (the error event). Use a longer confirm duration
# to verify no additional data events arrive.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

# The single document should be an error event.
exec jq '[.hits.hits[]._source | select(.error != null)] | length' got_docs.json
stdout '^1$'

# The error message should carry the Cisco Duo API error, unparsed.
exec jq -r '.hits.hits[0]._source.error.message' got_docs.json
stdout 'Invalid signature in request credentials'

# The terminate_collector_error branch must prevent any pipeline failure.
exec jq '[.hits.hits[]._source | select(.event.kind == "pipeline_error")] | length' got_docs.json
stdout '^0$'

# No auth data events should be present (every doc is an error event).
exec jq '[.hits.hits[]._source | select(.error == null)] | length' got_docs.json
stdout '^0$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down duo-mock

-- test_config.yaml --
input: cel
vars:
  hostname: http://duo-mock:8080
  integration_key: wrong-integration-key
  secret_key: wrong-secret-key
  interval: 30s
data_stream:
  vars:
    ignore_api_errors: false
    preserve_original_event: true
-- duo-mock/docker-compose.yml --
version: '2.3'
services:
  duo-mock:
    image: docker.elastic.co/observability/stream:v0.20.0
    hostname: duo-mock
    ports:
      - 8080
    environment:
      PORT: "8080"
    volumes:
      - ./config.yml:/config.yml
    command:
      - http-server
      - --addr=:8080
      - --config=/config.yml
-- duo-mock/config.yml --
rules:
  # Authentication endpoint: all requests return 401 with the Cisco Duo error
  # body returned for an invalid request signature (wrong integration/secret key).
  - path: /admin/v2/logs/authentication
    methods: [GET]
    responses:
      - status_code: 401
        headers:
          Content-Type:
            - "application/json"
        body: |-
          {"code": 40103, "message": "Invalid signature in request credentials", "stat": "FAIL"}
