# Verify that events from the threat data stream are routed to event-type-specific
# data streams via routing rules.
#
# The mock HTTP server returns one sample event in each of the four categories
# returned by the /v2/siem/all endpoint:
#   - clicksBlocked    → proofpoint_essentials.clicks_blocked (rerouted)
#   - clicksPermitted  → proofpoint_essentials.clicks_permitted (rerouted)
#   - messagesBlocked  → proofpoint_essentials.message_blocked (rerouted)
#   - messagesDelivered → proofpoint_essentials.message_delivered (rerouted)
#
# The sample data is drawn from the existing pipeline test fixtures and from
# the Proofpoint Essentials SIEM API documentation.

[!external_stack] skip 'Skipping external stack test.'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock HTTP server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} ppe-mock

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy pointing at the mock server.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml THREAT_DS

# Each event type should route to its own data stream.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 30s -timeout 5m logs-proofpoint_essentials.clicks_blocked-*
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 30s -timeout 5m logs-proofpoint_essentials.clicks_permitted-*
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 30s -timeout 5m logs-proofpoint_essentials.message_blocked-*
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 30s -timeout 5m logs-proofpoint_essentials.message_delivered-*

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${THREAT_DS}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down ppe-mock

-- test_config.yaml --
input: cel
vars:
  url: http://svc-ppe-mock:8080
  api_key: testkey
  api_key_secret: testsecret
data_stream:
  vars:
    initial_interval: 2h
    interval: 15s
    collect_customer_data: true
    collect_own_data: true
    preserve_original_event: false

-- ppe-mock/docker-compose.yml --
services:
  ppe-mock:
    image: docker.elastic.co/observability/stream:v0.19.0
    ports:
      - 8080
    environment:
      PORT: "8080"
    volumes:
      - ./config.yml:/config.yml
    healthcheck:
      test: ["CMD-SHELL", "wget --no-verbose --tries=1 --spider http://localhost:8080/health || exit 1"]
      interval: 5s
      timeout: 3s
      retries: 10
    command:
      - http-server
      - --addr=:8080
      - --config=/config.yml
