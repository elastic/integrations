# End-to-end fresh-install bootstrap test verifying #22740:
# - With zero metrics-chargeback.* and zero chargeback_conf_lookup* indices on a fresh stack,
#   chargeback_conf_lookup and chargeback_currency_lookup start cleanly (state=started, health=green)
#   and materialize empty lookup-mode destination indices + aliases on Checkpoint 1.
# - ES|QL LOOKUP JOIN queries against chargeback_lookup.chargeback_conf and
#   chargeback_lookup.chargeback_currency succeed immediately with default rates (0.85 EUR / €).
# - Once the first CEL/Agentless config_snapshot event arrives into metrics-chargeback.billing-*
#   after Checkpoint 1, Checkpoint 2 materializes _id=config, _id=config_onprem, and currency _id=config.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'
[!exec:sh] skip 'Skipping test requiring absent sh command'
[!exec:curl] skip 'Skipping test requiring absent curl command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Ensure a completely clean starting state with zero Chargeback indices or transforms.
exec sh scripts/cleanup.sh

# Install only the 0.7.1 package assets (no Agent policy yet -> zero metrics-chargeback.* indices).
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
stdout 'added package resources for '${PACKAGE_NAME}

# Verify fresh bootstrap materializes empty lookup indices before source indices exist,
# passes ES|QL LOOKUP JOIN, and syncs subsequent config_snapshot events cleanly.
exec sh scripts/verify_fresh_bootstrap.sh

# Final stack cleanup and restore the local package so subsequent test suites can run against it.
exec sh scripts/cleanup.sh
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

-- scripts/cleanup.sh --
#!/bin/sh
set -e
CA="${CONFIG_PROFILES}/${PROFILE}/certs/ca-cert.pem"
KBN="https://localhost:5601"
ES="https://localhost:9200"

POLS=$(curl -sS -u elastic:changeme --cacert "$CA" -H "kbn-xsrf: true" "$KBN/api/fleet/package_policies?perPage=100" | jq -r '[.items[]? | select(.package.name == "chargeback") | .id] | @json')
if [ "$POLS" != "[]" ] && [ -n "$POLS" ]; then
  curl -sS -u elastic:changeme --cacert "$CA" -H "kbn-xsrf: true" -H "Content-Type: application/json" -X POST "$KBN/api/fleet/package_policies/delete" -d "{\"packagePolicyIds\":$POLS}" >/dev/null || true
fi

VER=$(curl -sS -u elastic:changeme --cacert "$CA" -H "kbn-xsrf: true" "$KBN/api/fleet/epm/packages/chargeback" | jq -r 'if .item.status == "installed" then .item.version else "" end')
if [ -n "$VER" ]; then
  curl -sS -u elastic:changeme --cacert "$CA" -H "kbn-xsrf: true" -X DELETE "$KBN/api/fleet/epm/packages/chargeback-${VER}" >/dev/null || true
fi

for tid in $(curl -sS -u elastic:changeme --cacert "$CA" "$ES/_transform/logs-chargeback.*" | jq -r '.transforms[]?.id'); do
  curl -sS -u elastic:changeme --cacert "$CA" -X POST "$ES/_transform/${tid}/_stop?force=true&wait_for_completion=true" >/dev/null || true
  curl -sS -u elastic:changeme --cacert "$CA" -X DELETE "$ES/_transform/${tid}" >/dev/null || true
done

for ds in $(curl -sS -u elastic:changeme --cacert "$CA" "$ES/_data_stream/logs-chargeback.*,metrics-chargeback.*" | jq -r '.data_streams[]?.name'); do
  curl -sS -u elastic:changeme --cacert "$CA" -X DELETE "$ES/_data_stream/${ds}" >/dev/null || true
done
for idx in $(curl -sS -u elastic:changeme --cacert "$CA" "$ES/_cat/indices/chargeback_lookup.*,cluster_capacity_utilization_lookup,chargeback_conf_lookup*?format=json" | jq -r 'if type=="array" then .[]?.index else empty end'); do
  curl -sS -u elastic:changeme --cacert "$CA" -X DELETE "$ES/${idx}" >/dev/null || true
done

-- scripts/verify_fresh_bootstrap.sh --
#!/bin/sh
set -e
CA="${CONFIG_PROFILES}/${PROFILE}/certs/ca-cert.pem"
ES="https://localhost:9200"

# 1. Confirm zero Chargeback source indices exist on the fresh stack.
SRC_CNT=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/_cat/indices/metrics-chargeback.*,chargeback_conf_lookup*?format=json" | jq 'if type=="array" then length else 0 end')
[ "$SRC_CNT" = "0" ]

# 2. Confirm chargeback_conf_lookup, chargeback_currency_lookup, and billing_cluster_cost transforms
#    are installed at the active package version, then trigger _schedule_now for Checkpoint 1.
TID_CONF=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/_transform/logs-chargeback.chargeback_conf_lookup-default-*" | jq -r '.transforms[0].id // empty')
TID_CUR=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/_transform/logs-chargeback.chargeback_currency_lookup-default-*" | jq -r '.transforms[0].id // empty')
TID_BILL=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/_transform/logs-chargeback.billing_cluster_cost-default-*" | jq -r '.transforms[0].id // empty')
[ -n "$TID_CONF" ]
[ -n "$TID_CUR" ]
[ -n "$TID_BILL" ]
PKG_VER="${TID_CONF##*-default-}"
[ -n "$PKG_VER" ]

for tid in "$TID_CONF" "$TID_CUR" "$TID_BILL"; do
  curl -sSf -u elastic:changeme --cacert "$CA" -X POST "$ES/_transform/${tid}/_schedule_now" >/dev/null
done

i=0
boot_ok="0"
while [ "$i" -lt 30 ]; do
  s_conf=$(curl -sS -o /dev/null -w "%{http_code}" -u elastic:changeme --cacert "$CA" "$ES/_alias/chargeback_lookup.chargeback_conf")
  s_cur=$(curl -sS -o /dev/null -w "%{http_code}" -u elastic:changeme --cacert "$CA" "$ES/_alias/chargeback_lookup.chargeback_currency")
  s_bill=$(curl -sS -o /dev/null -w "%{http_code}" -u elastic:changeme --cacert "$CA" "$ES/_alias/chargeback_lookup.billing_cluster_cost")
  if [ "$s_conf" = "200" ] && [ "$s_cur" = "200" ] && [ "$s_bill" = "200" ]; then
    boot_ok="1"
    break
  fi
  i=$((i + 1))
  sleep 2
done
[ "$boot_ok" = "1" ]

# 3. Verify transform states are started + green and destinations are empty lookup-mode indices.
for tid in "$TID_CONF" "$TID_CUR"; do
  st=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/_transform/${tid}/_stats")
  printf '%s' "$st" | jq -e '.transforms[0].state == "started" and .transforms[0].health.status == "green"' >/dev/null
done

for pair in \
  "chargeback_lookup.dest_chargeback_conf-1:chargeback_lookup.chargeback_conf" \
  "chargeback_lookup.dest_chargeback_currency-1:chargeback_lookup.chargeback_currency"; do
  idx="${pair%%:*}"
  al="${pair##*:}"
  al_keys=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/_alias/${al}" | jq -c 'keys | sort')
  [ "$al_keys" = "[\"${idx}\"]" ]
  imode=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/${idx}/_settings" | jq -r --arg idx "$idx" '.[$idx].settings.index.mode')
  idyn=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/${idx}/_mapping" | jq -r --arg idx "$idx" '.[$idx].mappings.dynamic')
  cnt=$(curl -sSf -u elastic:changeme --cacert "$CA" "$ES/${idx}/_count" | jq -r '.count')
  [ "$imode" = "lookup" ]
  [ "$idyn" = "false" ]
  [ "$cnt" = "0" ]
done

# 4. Verify ES|QL LOOKUP JOIN works immediately on empty config lookups using default 0.85 EUR / €.
curl -sSf -u elastic:changeme --cacert "$CA" -H "Content-Type: application/json" -X PUT "$ES/chargeback_lookup.dest_billing_cluster_cost-1/_doc/fresh_bill_1?refresh=true" -d '{"@timestamp":"2026-01-15T00:00:00Z","composite_key":"2026-01-15_dep-fresh","deployment_id":"dep-fresh","deployment_name":"fresh-dep","deployment_group":"team-fresh","cost_category":"data_tier","is_onprem":false,"total_chargeable_units":100.0}' >/dev/null

jq -n --arg q 'FROM chargeback_lookup.billing_cluster_cost
| WHERE @timestamp IS NOT NULL AND deployment_id == "dep-fresh"
| EVAL ts = @timestamp, cloud_cfg_key = "chargeback_config"
| LOOKUP JOIN chargeback_lookup.chargeback_conf ON cloud_cfg_key == config_join_key AND ts >= conf_start_date AND ts <= conf_end_date
| EVAL cloud_rate = conf_chargeable_unit_rate, cloud_unit = conf_chargeable_unit_rate_unit
| DROP config_join_key, conf_start_date, conf_end_date, conf_chargeable_unit_rate, conf_chargeable_unit_rate_unit
| LOOKUP JOIN chargeback_lookup.chargeback_currency ON cloud_cfg_key == config_join_key AND ts >= conf_start_date AND ts <= conf_end_date
| EVAL cloud_sym = conf_currency_symbol, cloud_sym_unit = conf_currency_unit
| DROP cloud_cfg_key, config_join_key, conf_start_date, conf_end_date, conf_currency_symbol, conf_currency_unit
| EVAL _c_unit = CASE(LENGTH(TRIM(cloud_unit)) > 0, TO_UPPER(TRIM(cloud_unit)), null),
       _c_sym_u = CASE(LENGTH(TRIM(cloud_sym_unit)) > 0, TO_UPPER(TRIM(cloud_sym_unit)), null)
| EVAL _c_sym = CASE(LENGTH(TRIM(cloud_sym)) > 0 AND (_c_sym_u IS NULL OR _c_sym_u == COALESCE(_c_unit, "EUR")), TRIM(cloud_sym), null),
       rate = COALESCE(cloud_rate, 0.85),
       unit = COALESCE(_c_unit, "EUR")
| EVAL _def_sym = CASE(unit == "USD", "$", unit == "GBP", "£", unit == "ZAR", "R", "€")
| EVAL sym = CASE(_c_sym IS NOT NULL, _c_sym, _def_sym),
       cost = COALESCE(total_chargeable_units, 0.0) * rate
| STATS spend = ROUND(SUM(cost), 2), sym = MAX(sym) BY deployment_id, unit
| KEEP deployment_id, spend, unit, sym' '{"query": $q}' > fresh_spend.json

curl -sSf -u elastic:changeme --cacert "$CA" -H "Content-Type: application/json" -X POST "$ES/_query?format=json" -d @fresh_spend.json > fresh_spend_res.json
jq -e '.values | length == 1 and (.[0][1] == 85 or .[0][1] == 85.0) and .[0][2] == "EUR" and .[0][3] == "€"' fresh_spend_res.json >/dev/null

# 5. Simulate the first CEL/Agentless config_snapshot arriving after Checkpoint 1 and verify
#    that subsequent checkpoints materialize _id=config, _id=config_onprem, and currency _id=config.
curl -sSf -u elastic:changeme --cacert "$CA" -H "Content-Type: application/json" -X POST "$ES/metrics-chargeback.billing-default/_doc?pipeline=metrics-chargeback.billing-${PKG_VER}&refresh=true" -d '{"@timestamp":"2026-01-15T00:00:00Z","data_stream":{"type":"metrics","dataset":"chargeback.billing","namespace":"default"},"doc_type":"config_snapshot","conf_sync_enabled":true,"conf_schema_version":2,"conf_policy_id":"cel-chargeback.billing-fresh-1","conf_chargeable_unit_rate":1.20,"conf_onprem_chargeable_unit_rate":0.0012,"conf_chargeable_unit_rate_unit":"USD","conf_currency_unit":"USD","conf_currency_symbol":""}' >/dev/null

sleep 65
curl -sSf -u elastic:changeme --cacert "$CA" -X POST "$ES/_transform/${TID_CONF}/_schedule_now" >/dev/null
curl -sSf -u elastic:changeme --cacert "$CA" -X POST "$ES/_transform/${TID_CUR}/_schedule_now" >/dev/null

i=0
synced_ok="0"
while [ "$i" -lt 30 ]; do
  ok_cloud="0"
  ok_onprem="0"
  ok_cur="0"
  if curl -sSf -u elastic:changeme --cacert "$CA" "$ES/chargeback_lookup.dest_chargeback_conf-1/_doc/config" 2>/dev/null | jq -e '.found == true and ._source.conf_chargeable_unit_rate == 1.2' >/dev/null 2>&1; then
    ok_cloud="1"
  fi
  if curl -sSf -u elastic:changeme --cacert "$CA" "$ES/chargeback_lookup.dest_chargeback_conf-1/_doc/config_onprem" 2>/dev/null | jq -e '.found == true and ._source.conf_chargeable_unit_rate == 0.0012' >/dev/null 2>&1; then
    ok_onprem="1"
  fi
  if curl -sSf -u elastic:changeme --cacert "$CA" "$ES/chargeback_lookup.dest_chargeback_currency-1/_doc/config" 2>/dev/null | jq -e '.found == true and ._source.conf_currency_symbol == "$"' >/dev/null 2>&1; then
    ok_cur="1"
  fi
  if [ "$ok_cloud" = "1" ] && [ "$ok_onprem" = "1" ] && [ "$ok_cur" = "1" ]; then
    synced_ok="1"
    break
  fi
  i=$((i + 1))
  sleep 2
done
[ "$synced_ok" = "1" ]

curl -sSf -u elastic:changeme --cacert "$CA" -X POST "$ES/chargeback_lookup.dest_chargeback_conf-1,chargeback_lookup.dest_chargeback_currency-1/_refresh" >/dev/null
curl -sSf -u elastic:changeme --cacert "$CA" -H "Content-Type: application/json" -X POST "$ES/_query?format=json" -d @fresh_spend.json > fresh_spend_synced.json
jq -e '.values | length == 1 and (.[0][1] == 120 or .[0][1] == 120.0) and .[0][2] == "USD" and .[0][3] == "$"' fresh_spend_synced.json >/dev/null
