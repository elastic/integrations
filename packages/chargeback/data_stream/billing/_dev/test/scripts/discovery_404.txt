# Test that HTTP 404 from /api/v1/deployments/_search fails closed (ok=false, retryable=false)
# instead of silently returning empty targets, emits error.code == "404", and blocks
# unattributed ECH deployment billing line items.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} cb-mock

add_package -profile ${CONFIG_PROFILES}/${PROFILE}

add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Expect 2 documents:
#   1 config_snapshot (conf_sync_enabled=false)
#   1 error document with error.code == "404"
# Unattributed ECH billing_line_item documents must be blocked (0 emitted).
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '[.hits.hits[]._source | select(.error.code == "404" and .event.ingested != null)] | length' got_docs.json
stdout '^1$'

exec jq '[.hits.hits[]._source | select(.ess.billing.deployment_id != null)] | length' got_docs.json
stdout '^0$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down cb-mock
