# Test deployment_api_keys auth fallback and strict short-circuiting:
# - dep-ech-01: first key returns 401 -> falls back to second key (200), emitting node_sample.
# - hub-cluster-01: first key returns 500 -> short-circuits immediately (does NOT try fallback 200 key), emitting error.code == "500".

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} cb-mock

add_package -profile ${CONFIG_PROFILES}/${PROFILE}

add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Expect 3 documents:
#   1 config_snapshot
#   1 node_sample (dep-ech-01 after 401 -> 200 fallback)
#   1 error (hub-cluster-01 500 short-circuit without retrying fallback 200 key)
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '[.hits.hits[]._source | select(.elasticsearch.cluster.name == "dep-ech-01" and .elasticsearch.node.name == "instance-0000000001")] | length' got_docs.json
stdout '^1$'

exec jq '[.hits.hits[]._source | select(.error.code == "500" and (.error.message | contains("/hub/_nodes")) and .event.ingested != null)] | length' got_docs.json
stdout '^1$'

exec jq '[.hits.hits[]._source | select(.elasticsearch.cluster.name == "hub-cluster-01")] | length' got_docs.json
stdout '^0$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down cb-mock
