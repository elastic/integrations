# Test that _shards.failed > 0 triggers Soft-Fail + partial_shard_stats tagging
# instead of blocking all index deltas, while indices with zero responding shards
# (partial_empty) are filtered out so existing baselines are not zeroed.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} cb-mock

add_package -profile ${CONFIG_PROFILES}/${PROFILE}

add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Expect 3 documents:
#   1 config_snapshot (conf_sync_enabled=false)
#   1 partial_shard_stats diagnostic error document
#   1 index_usage document for orders-2026.01.01 (broken-2026.01.01 filtered via partial_empty)
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '[.hits.hits[]._source | select(.error.code == "partial_shard_stats")] | length' got_docs.json
stdout '^1$'

exec jq '[.hits.hits[]._source | select(.elasticsearch.index.name == "orders-2026.01.01" and .elasticsearch.index.partial_shard_stats == true)] | length' got_docs.json
stdout '^1$'

exec jq '[.hits.hits[]._source | select(.elasticsearch.index.name == "broken-2026.01.01")] | length' got_docs.json
stdout '^0$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down cb-mock
