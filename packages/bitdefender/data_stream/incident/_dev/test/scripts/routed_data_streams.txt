# Verify end-to-end delivery of both parent incidents and routed child alerts
# in bitdefender.incident_alert within the configured namespace.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} bitdefender-incident-script-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Verify both parent incidents land in the policy's bitdefender.incident stream with normalized risk_score and severity.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 10s -timeout 3m ${DATA_STREAM_NAME}
cp stdout parent_docs.json
exec jq -r '[.hits.hits[]._source["event.id"]] | sort | join(",")' parent_docs.json
stdout '^65f04c2e1234567890abc00c,65f04c2e1234567890abc00e$'
exec jq -r '[.hits.hits[]._source | {id: .["event.id"], risk: .["event.risk_score"], sev: .["event.severity"]}] | sort_by(.id) | map("\(.id):\(.risk):\(.sev)") | join(",")' parent_docs.json
stdout '^65f04c2e1234567890abc00c:55(\.0)?:73,65f04c2e1234567890abc00e:90(\.0)?:99$'
exec jq '[.hits.hits[]._source | to_entries[] | select(.key | startswith("bitdefender.incident_alert") or startswith("routing"))] | length' parent_docs.json
stdout '^0$'
exec jq -r '.hits.hits[0]._source["data_stream.namespace"]' parent_docs.json
cp stdout expected_ns.txt

# Verify both embedded child alerts are routed to bitdefender.incident_alert in the exact same namespace.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 10s -timeout 3m logs-bitdefender.incident_alert-*
cp stdout alert_docs.json
exec jq -r --rawfile ns expected_ns.txt '($ns | sub("\n$"; "")) as $target_ns | [.hits.hits[]._source | select(.["data_stream.namespace"] == $target_ns) | .["event.id"]] | sort | join(",")' alert_docs.json
stdout '^75a1b2c3d4e5f67890abc001,75a1b2c3d4e5f67890abc002$'
exec jq -r --rawfile ns expected_ns.txt '($ns | sub("\n$"; "")) as $target_ns | [.hits.hits[]._source | select(.["data_stream.namespace"] == $target_ns) | {id: .["event.id"], parent: .["bitdefender.incident_alert.incident.incidentId"], ds: .["data_stream.dataset"], risk: .["event.risk_score"], sev: .["event.severity"]}] | sort_by(.id) | map("\(.id):\(.parent):\(.ds):\(.risk):\(.sev)") | join(",")' alert_docs.json
stdout '^75a1b2c3d4e5f67890abc001:65f04c2e1234567890abc00c:bitdefender.incident_alert:55(\.0)?:73,75a1b2c3d4e5f67890abc002:65f04c2e1234567890abc00e:bitdefender.incident_alert:90(\.0)?:99$'
exec jq -r --rawfile ns expected_ns.txt '($ns | sub("\n$"; "")) as $target_ns | .hits.hits[]._source | select(.["data_stream.namespace"] == $target_ns and .["event.id"] == "75a1b2c3d4e5f67890abc001") | "\(.["rule.name"])|\(.["process.name"])|\(.["registry.path"])"' alert_docs.json
stdout '^Synthetic Process Detection\|example.exe\|HKLM\\Software\\Example$'
exec jq -r --rawfile ns expected_ns.txt '($ns | sub("\n$"; "")) as $target_ns | .hits.hits[]._source | select(.["data_stream.namespace"] == $target_ns and .["event.id"] == "75a1b2c3d4e5f67890abc002") | "\(.["rule.name"])|\(.["threat.tactic.id"][0])"' alert_docs.json
stdout '^SyntheticCloudActivity\|TA0002$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down bitdefender-incident-script-mock
