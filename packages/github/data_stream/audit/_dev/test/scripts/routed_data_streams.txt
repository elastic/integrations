# Verify that GitHub audit log events with event.action api.request are routed to
# github.api_requests when route_api_request_events is enabled on a streaming input.
#
# The GCS mock serves two NDJSON events: one api.request and one repo.create.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} github-audit-gcs-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 15s -timeout 5m logs-github.api_requests-*
cp stdout api_request_docs.json
exec jq '[.hits.hits[]._source | select(.event.action == "api.request")] | length' api_request_docs.json
stdout '^1$'
exec jq -r '[.hits.hits[]._source.event.id] | unique[]' api_request_docs.json
stdout 'Zt3XoecTsz6kAS-87bAEzQ'

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -confirm 15s -timeout 5m logs-github.audit-*
cp stdout audit_docs.json
exec jq '[.hits.hits[]._source | select(.event.action == "api.request")] | length' audit_docs.json
stdout '^0$'
exec jq '[.hits.hits[]._source | select(.event.action == "repo.create")] | length' audit_docs.json
stdout '^1$'
exec jq -r '[.hits.hits[]._source.event.id] | unique[]' audit_docs.json
stdout 'Nm4EoecTsz6kAS-87bAEfA'

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 15s -timeout 5m logs-github.audit-*,logs-github.api_requests-*
cp stdout all_docs.json
exec jq '.hits.total.value' all_docs.json
stdout '^2$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down github-audit-gcs-mock

-- test_config.yaml --
input: gcs
data_stream:
  vars:
    project_id: fake-gcs-project
    alternative_host: "http://svc-github-audit-gcs-mock:4443"
    service_account_key: '{"type":"service_account","project_id":"fake-gcs-project"}'
    number_of_workers: 1
    poll: true
    poll_interval: 15s
    buckets: |
      - name: routing-test-bucket
    route_api_request_events: true
    preserve_original_event: false
-- github-audit-gcs-mock/docker-compose.yml --

  github-audit-gcs-mock:
    image: shourieg/gcs-mock-service:latest

    ports:
      - '4443/tcp'
    volumes:
      - ./files/manifest.yml:/app/manifest.yml

    healthcheck:
      test: 'wget --no-verbose --tries=1 --spider http://localhost:4443/health || exit 1'
      interval: 10s
      timeout: 5s
    command: ['./gcs-mock-service', '--manifest', '/app/manifest.yml']
