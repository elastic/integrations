# Test the windows that the unified_alert data stream queries.
#
# The mock returns the same alert to every request and logs every request
# body. After a few polls the test stops the mock and checks the
# dateTimeRange of the logged requests:
#   - every window has an inclusive start;
#   - the detectedAt and updatedAt requests of a poll use the same window;
#   - the first window spans the initial interval;
#   - every later window starts where the previous one ended;
#   - every window ends about the delay (5m) before the request.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

# Connect to the running stack and install an agent.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock SentinelOne API server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} s1-mock

# Add the package resources and the data stream policy.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -timeout 5m ${DATA_STREAM_NAME}

# Let at least three more polls run.
sleep 40s

# Stopping the mock prints its log.
docker_down s1-mock
cp stdout mock.log

exec jq -n -R -c -f requests.jq mock.log
cp stdout requests.ndjson

# check.jq lists the checks that failed.
exec jq -s -c -f check.jq requests.ndjson
stdout '^\[\]$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m

-- test_config.yaml --
input: cel
vars:
  url: http://s1-mock:8080
  api_token: test-api-token
data_stream:
  vars:
    interval: 10s
    initial_interval: 1h
    delay: 5m
    batch_size: 10
    preserve_original_event: true
    enable_request_tracer: false
-- requests.jq --
# Print one object per request in the mock's log, with the time the
# request was received (ms since the epoch) and the window it queried.
def epoch_ms:
  capture("^(?<s>[^.]+)[.](?<ms>[0-9]{3})(?<z>Z|[+-][0-9]{4})$")
  | (.s + "Z" | fromdateiso8601) * 1000 + (.ms | tonumber)
    - (if .z == "Z" then 0 else
        (if .z[0:1] == "-" then -1 else 1 end)
        * ((.z[1:3] | tonumber) * 3600000 + (.z[3:5] | tonumber) * 60000)
      end);

inputs
| sub("^[^{]*"; "")
| fromjson?
| select((.msg // "") | test("^Rule #[0-9]+ matched"))
| (.msg | sub("^.*?Request Body: "; "") | fromjson | .variables.filters[0]) as $f
| {
    received: (.ts | epoch_ms),
    field: $f.fieldId,
    start: $f.dateTimeRange.start,
    end: $f.dateTimeRange.end,
    inclusive: $f.dateTimeRange.startInclusive
  }
-- check.jq --
# Drop a poll that was cut off by stopping the mock.
(if length % 2 == 1 then .[:-1] else . end) as $reqs
| [range(0; $reqs | length; 2) as $i | $reqs[$i:$i+2]] as $polls
| [
    if ($polls | length) < 3 then "fewer than 3 polls: \($polls | length)" else empty end,
    ($reqs[] | select(.inclusive != true) | "exclusive start: \(.)"),
    ($reqs[] | select(.received - .end < 295000 or .received - .end > 315000) | "window does not end 5m before the request: \(.)"),
    ($polls[] | select(.[0].field != "detectedAt" or .[1].field != "updatedAt" or .[0].start != .[1].start or .[0].end != .[1].end) | "phases do not share a window: \(.)"),
    ($polls[0][0] | select(.end - .start != 3600000) | "first window does not span the initial interval: \(.)"),
    (range(1; $polls | length) as $i | $polls[$i][0] | select(.start != $polls[$i - 1][0].end) | "window does not start at the previous end: \(.)")
  ]
