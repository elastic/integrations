# Test that upgrading from the latest released version keeps the
# unified_alert data stream's windows contiguous, apart from going back
# the delay once.
#
# The released version's windows end at the poll time, and its cursor
# holds the end of the last window but no queried_at. The first window
# after the upgrade must start the delay (5m) before the end of one of
# the released version's windows, so that it covers alerts that were not
# yet searchable then. Later windows must start where the previous one
# ended.
#
# The released version queries with an exclusive start and the upgraded
# version with an inclusive start, which tells their requests apart in
# the mock's log.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'
[!env:LATEST_EPR_VERSION] skip 'no EPR release to upgrade from'
[!is_latest_version] skip 'dev version is not newer than latest EPR release'
[breaking_change] skip 'Cannot upgrade across breaking change.'

# Connect to the running stack and install an agent.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock SentinelOne API server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} s1-mock

# Install the released version and create a policy from its manifests.
install_package_from_registry -profile ${CONFIG_PROFILES}/${PROFILE} ${PACKAGE_NAME} ${LATEST_EPR_VERSION}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} -version ${LATEST_EPR_VERSION} test_config.yaml DATA_STREAM_NAME

get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 1 -timeout 5m ${DATA_STREAM_NAME}

# Let the released version complete a few polls and save its cursor.
sleep 30s

# Upgrade to the local version.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
upgrade_package_latest -profile ${CONFIG_PROFILES}/${PROFILE}
stdout 'upgraded package '${PACKAGE_NAME}

# Wait for the agent to apply the upgraded policy and poll a few times.
sleep 60s

# Stopping the mock prints its log.
docker_down s1-mock
cp stdout mock.log

exec jq -n -R -c -f requests.jq mock.log
cp stdout requests.ndjson

# check.jq lists the checks that failed.
exec jq -s -c -f check.jq requests.ndjson
stdout '^\[\]$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m

-- test_config.yaml --
input: cel
vars:
  url: http://s1-mock:8080
  api_token: test-api-token
data_stream:
  vars:
    interval: 10s
    initial_interval: 1h
    batch_size: 10
    preserve_original_event: true
    enable_request_tracer: false
-- requests.jq --
# Print one object per request in the mock's log, with the time the
# request was received (ms since the epoch) and the window it queried.
def epoch_ms:
  capture("^(?<s>[^.]+)[.](?<ms>[0-9]{3})(?<z>Z|[+-][0-9]{4})$")
  | (.s + "Z" | fromdateiso8601) * 1000 + (.ms | tonumber)
    - (if .z == "Z" then 0 else
        (if .z[0:1] == "-" then -1 else 1 end)
        * ((.z[1:3] | tonumber) * 3600000 + (.z[3:5] | tonumber) * 60000)
      end);

inputs
| sub("^[^{]*"; "")
| fromjson?
| select((.msg // "") | test("^Rule #[0-9]+ matched"))
| (.msg | sub("^.*?Request Body: "; "") | fromjson | .variables.filters[0]) as $f
| {
    received: (.ts | epoch_ms),
    field: $f.fieldId,
    start: $f.dateTimeRange.start,
    end: $f.dateTimeRange.end,
    inclusive: $f.dateTimeRange.startInclusive
  }
-- check.jq --
[.[] | select(.inclusive != true)] as $released
# Drop a poll that was cut off by stopping the mock.
| ([.[] | select(.inclusive == true)] | if length % 2 == 1 then .[:-1] else . end) as $upgraded
| [range(0; $upgraded | length; 2) as $i | $upgraded[$i:$i+2]] as $polls
| [
    if ($released | length) == 0 then "no requests from the released version" else empty end,
    if ($polls | length) < 2 then "fewer than 2 polls after the upgrade: \($polls | length)" else empty end,
    ($polls[0][0] | select(.start + 300000 as $end | any($released[]; .end == $end) | not) | "first window after the upgrade does not start 5m before a released window end: \(.)"),
    ($upgraded[] | select(.received - .end < 295000 or .received - .end > 315000) | "window does not end 5m before the request: \(.)"),
    ($polls[] | select(.[0].field != "detectedAt" or .[1].field != "updatedAt" or .[0].start != .[1].start or .[0].end != .[1].end) | "phases do not share a window: \(.)"),
    (range(1; $polls | length) as $i | $polls[$i][0] | select(.start != $polls[$i - 1][0].end) | "window does not start at the previous end: \(.)")
  ]
