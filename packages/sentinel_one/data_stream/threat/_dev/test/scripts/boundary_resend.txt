# Test that the threat data stream does not re-send the most recently updated
# threat on every poll.
#
# SentinelOne's updatedAt__gte filter is inclusive, so the threat that set the
# cursor is returned again by every later poll. The program must not index it
# again; otherwise it is re-indexed after each backing index rollover and the
# prebuilt detection rule alerts on it again.
#
# The mock answers by the updatedAt__gte value the agent sends:
#   initial poll          -> threat 1 (updatedAt 10:01)
#   updatedAt__gte 10:01  -> threat 1, threat 1, then threat 1 and threat 3
#   updatedAt__gte 10:30  -> threat 3
# Threat 3 only appears on the third poll after threat 1, so having it indexed
# shows that at least two polls returned only the boundary threat. The test
# asserts both threats exist and that no write was rejected as a version
# conflict.

# The mock threats carry a top-level "id", as the real API does. The ingest
# pipeline builds _id from it, so a re-sent threat is rejected as a version
# conflict inside one backing index. The document count cannot show a re-send,
# so the tests assert on the data stream's version conflict counter.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'
[!exec:curl] skip 'Skipping test requiring absent curl command'

# Connect to the running stack and install an agent.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock SentinelOne API server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} s1-mock

# Add the package resources and the data stream policy.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Wait for both threats, then confirm that nothing else arrives.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 30s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '[.hits.hits[]._source.sentinel_one.threat.id] | sort | join(",")' got_docs.json
stdout '"1000000000000000001,1000000000000000003"'

exec curl -s -u elastic:changeme --cacert ${CONFIG_PROFILES}/${PROFILE}/certs/ca-cert.pem -X GET 'https://localhost:9200/'${DATA_STREAM_NAME}'/_stats/indexing'
cp stdout got_stats.json

# Version conflicts count writes rejected because the document already exists
# in the backing index, so each re-sent threat adds one.
exec jq '._all.primaries.indexing.index_failed_due_to_version_conflict' got_stats.json
stdout '^0$'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down s1-mock

  url: http://s1-mock:8080
  api_token: test-api-token
data_stream:
  vars:
    interval: 10s
    initial_interval: 1h
    preserve_original_event: true
    enable_request_tracer: false
