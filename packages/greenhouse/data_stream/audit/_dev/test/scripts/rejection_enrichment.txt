# Verify the exact values produced by the rejection enrichment, end to end, against the
# package mock in _dev/deploy/docker (10 audit events over two pages). The four rejections:
#   - 7890dID: its RejectionDetails sibling is on page 2 (cross-page correlation) and its
#     application has two rejection_details records (a re-rejection); the record whose id
#     matches the RejectionDetails event (221864428002, reason 555) must be used, not the
#     older one (reason 556).
#   - 7891eID: two RejectionDetails share the request.id (bulk rejection) -> ambiguous error.
#   - 7892fID: no RejectionDetails sibling -> held for one window, then a "no RejectionDetails"
#     error on the next poll.
#   - 7893gID: /v3/rejection_reasons/999 returns 404 -> reason.error; notes still attached.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Reuse the package's system test mock. The directory name must match the compose service
# name (greenhouse), which is also the hostname the policy below points at.
exec ln -s ${PACKAGE_ROOT}/_dev/deploy/docker greenhouse
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} greenhouse

add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# All 10 events; the held rejection only arrives on the second poll. Hold steady to verify
# that the events re-read by the overlapping second window do not create extra documents.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 10 -confirm 30s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq '.hits.total.value' got_docs.json
stdout '^10$'

# 7890dID: correlated across pages and enriched from the matching rejection_details record.
exec jq -r '.hits.hits[]._source | select(.event.id == "7890dID" and .greenhouse.audit.event.target_type == "Candidate or Prospect rejected") | .greenhouse.audit.event.rejection | "\(.application_id)|\(.reason.id)|\(.reason.name)|\(.reason.type)|\(.notes)|\(.rejected_at)|\(.error)"' got_docs.json
stdout '^294326368002\|555\|Not a fit\|We rejected them\|Candidate lacked required experience with distributed systems\.\|2023-06-05T09:00:00\.000Z\|null$'

exec jq -r '.hits.hits[]._source | select(.event.id == "7890dID" and .greenhouse.audit.event.target_type == "Candidate or Prospect rejected") | .event.reason' got_docs.json
stdout '^Not a fit$'

# The older record's reason and note must not appear on any document.
exec jq -r '[.hits.hits[]._source.greenhouse.audit.event.rejection? | select(. != null) | (.reason.name // ""), (.notes // "")] | join("|")' got_docs.json
! stdout 'Position closed'
! stdout 'Older rejection'

# 7891eID: ambiguous bulk rejection.
exec jq -r '.hits.hits[]._source | select(.event.id == "7891eID" and .greenhouse.audit.event.target_type == "Candidate or Prospect rejected") | .greenhouse.audit.event.rejection | "\(.error)|\(.ambiguous_application_ids | sort | join(","))|\(.application_id)"' got_docs.json
stdout '^ambiguous: 2 RejectionDetails events share request\.id 7891eID \(likely a bulk rejection\)\|111111,222222\|null$'

# 7892fID: no sibling anywhere; resolved with an error after being held for one window.
exec jq -r '.hits.hits[]._source | select(.event.id == "7892fID") | .greenhouse.audit.event.rejection | "\(.error)|\(.application_id)"' got_docs.json
stdout '^no RejectionDetails audit event found sharing request\.id 7892fID\|null$'

# 7893gID: reason lookup failed, notes and rejected_at still attached.
exec jq -r '.hits.hits[]._source | select(.event.id == "7893gID" and .greenhouse.audit.event.target_type == "Candidate or Prospect rejected") | .greenhouse.audit.event.rejection | "\(.application_id)|\(.reason.error)|\(.reason.name)|\(.notes)|\(.rejected_at)"' got_docs.json
stdout '^300000000001\|rejection_reason fetch failed: HTTP 404\|null\|Interview feedback was negative\.\|2023-06-05T09:15:00\.000Z$'

# Exactly the three failed enrichments are tagged.
exec jq -r '[.hits.hits[]._source | select(.tags | index("greenhouse-rejection-enrichment-failed")) | .event.id] | sort | join(",")' got_docs.json
stdout '^7891eID,7892fID,7893gID$'

# The audit event itself is emitted byte-for-byte: numeric ids stay numbers in event.original,
# including on enriched events and on the passthrough RejectionDetails event.
exec jq -r '.hits.hits[]._source | select(.event.id == "7890dID" and .greenhouse.audit.event.target_type == "Candidate or Prospect rejected") | .event.original | fromjson | "\(.performer.id)|\(.organization_id)|\(.event | has("rejection"))"' got_docs.json
stdout '^22222\|123\|false$'

exec jq -r '.hits.hits[]._source | select(.event.id == "7890dID" and .greenhouse.audit.event.target_type == "RejectionDetails") | .event.original' got_docs.json
stdout '"target_id":221864428002,'
stdout '"application_id":294326368002,'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down greenhouse
