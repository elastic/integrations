# Test that replacing the agent (full re-enrollment, no cursor carried over)
# does not lose incidents or their alerts, and that repeated delivery of the
# same incident does not create duplicates or collapse its alerts.
#
# The incident data stream pages the Graph Security API with a
# `lastUpdateDateTime ge <cursor>` filter and derives the Elasticsearch document
# `_id` from a fingerprint of the incident (id + lastUpdateDateTime +
# incidentWebUrl + createdDateTime) *and* the alert (alerts.id +
# alerts.lastUpdateDateTime). Both halves matter:
#
#   - The incident half makes re-delivery idempotent, exactly as for the alert
#     data stream: the `ge` cursor re-includes the boundary incident on each
#     poll, and the same incident always maps to the same `_id`.
#   - The alert half keeps the fan-out distinct. With include_alerts enabled the
#     input splits body.alerts with keep_parent, so one incident becomes one
#     document per alert, and every one of those documents carries identical
#     incident-level fields. Only alerts.id and alerts.lastUpdateDateTime
#     separate them. If they were ever dropped from the fingerprint, an
#     incident's alerts would overwrite each other down to a single document —
#     silent loss rather than duplication.
#
# Phase 1: the mock returns the same two incidents on every poll. incident-001
# carries three alerts and incident-002 carries one, so four documents are
# expected. The test asserts exactly four land, which proves both that repeated
# delivery de-duplicates and that the three alerts of incident-001 stay
# distinct.
#
# Phase 2: the agent is destroyed and a fresh one is enrolled with no cursor
# state, so it re-collects from `initial_interval`. The mock is swapped so that
# incident-001 has aged out of the window, incident-002 is re-collected
# unchanged, and incident-003 is new with two alerts. Each install_agent creates
# a new Fleet policy with a unique namespace, so phase 2 writes to a separate
# data stream. The union across both phases must contain all six distinct
# incident/alert pairs, confirming the agent replacement lost nothing.
#
# NOTE: As for the alert data stream, this does NOT test mid-drain cursor
# consistency (the cursor advancing ahead of documents that have not yet been
# indexed). That needs the agent container killed mid-poll with its cursor
# preserved across restart, which the script test infrastructure does not
# support (uninstall_agent destroys the enrollment and its cursor).

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Start the mock Microsoft Graph Security API server (phase 1: incidents
# 001-002, four alerts between them).
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} m365d-mock

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy and start collection.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Wait for exactly 4 documents: three alerts from incident-001 and one from
# incident-002. The mock returns the same incidents on every poll, and interval
# is 10s, so -confirm 25s spans at least two further polls and proves the
# fingerprint `_id` de-duplicates repeated deliveries rather than inflating the
# count.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 4 -confirm 25s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_phase1.json

# Verify the four documents are the expected incident/alert pairs. Three of them
# share incident-001, which is the fan-out the alert half of the fingerprint
# keeps distinct. The incident id is read from event.id because the pipeline
# removes m365_defender.incident.id as an ECS duplicate unless
# preserve_duplicate_custom_fields is set.
exec jq '[.hits.hits[]._source | .event.id + "/" + .m365_defender.incident.alert.id] | sort | join(",")' got_docs_phase1.json
stdout '"incident-001/alert-001,incident-001/alert-002,incident-001/alert-003,incident-002/alert-004"'

# Only two distinct incidents produced those four documents.
exec jq '[.hits.hits[]._source.event.id] | unique | length' got_docs_phase1.json
stdout '^2$'

# --- Simulate agent replacement (node rotation, pod eviction) ---
# Destroy the agent entirely. The new agent will have no cursor state and will
# re-collect from initial_interval.
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down m365d-mock

# Swap the mock to phase 2: incident-001 aged out, incident-003 added with two
# alerts, incident-002 re-collected unchanged.
cp m365d-mock/config-phase2.yml m365d-mock/config.yml

# Install the agent first so NETWORK_NAME is refreshed, then start the mock on
# the agent's network so DNS resolution works.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} m365d-mock

# Re-assign the package policy. This sets DATA_STREAM_NAME to the new data
# stream (different namespace from phase 1).
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Wait for exactly 3 phase 2 documents in the new data stream.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 25s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs_phase2.json

# Phase 2 must contain the re-collected incident-002 alert plus both alerts of
# the new incident-003.
exec jq '[.hits.hits[]._source | .event.id + "/" + .m365_defender.incident.alert.id] | sort | join(",")' got_docs_phase2.json
stdout '"incident-002/alert-004,incident-003/alert-005,incident-003/alert-006"'

# Combine both phases and verify all six distinct incident/alert pairs are
# present: incident-001's three alerts preserved from phase 1, incident-002's
# alert re-collected, and incident-003's two alerts new in phase 2.
exec jq -s '[.[].hits.hits[]._source | .event.id + "/" + .m365_defender.incident.alert.id] | unique | sort | join(",")' got_docs_phase1.json got_docs_phase2.json
stdout '"incident-001/alert-001,incident-001/alert-002,incident-001/alert-003,incident-002/alert-004,incident-003/alert-005,incident-003/alert-006"'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down m365d-mock
