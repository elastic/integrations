# The Audit API rejects the stored cursorRef (403). The CEL program indexes one
# error document, drops cursor.ref, keeps cursor.last_timestamp, and on the
# next poll creates a new query whose dateFrom is the last event time.
#
# Request flow (interval 5s, page_size 3):
#   poll 1: token -> createQuery -> cursor-ref-query
#           results(cursor-ref-query)   -> 1 event (2099-11-03 14:42:42.914), cursor-ref-expired
#           results(cursor-ref-expired) -> 403, error document, cursor.ref dropped
#   poll 2: createQuery with dateFrom "2099-11-03 14:42:42" -> cursor-ref-resumed
#           results(cursor-ref-resumed)   -> 1 new event, cursor-ref-resumed-2
#           results(cursor-ref-resumed-2) -> 0 events (caught up)
#
# The createQuery rule for cursor-ref-resumed matches only the exact dateFrom.
# Any other second createQuery gets cursor-ref-wrong-resume, whose page has a
# guard event, so a wrong resume time fails the event ID check.
#
# The event timestamps are in 2099 because the program clamps dateFrom to at
# most 168h before now. A fixed past time would stop matching after 7 days.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# The token URL is built from identity_id, so the OAuth mock uses the host name
# elastic-audit-test.id.cyberark.cloud over HTTPS. The policy trusts its
# self-signed certificate through the ssl var.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} audit-oauth
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} audit-mock

add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# 2 audit events + 1 cursor-rejection error document, and no more.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json
exec jq '.hits.total.value' got_docs.json
stdout '^3$'

# Exactly one error document, from the rejected cursor.
exec jq '[.hits.hits[]._source | select(.error.message != null)] | length' got_docs.json
stdout '^1$'
exec jq -r '[.hits.hits[]._source | select(.error.message != null) | .error.message] | flatten | .[]' got_docs.json
stdout 'POST /api/audits/stream/results: cursor rejected, a new query starts on the next poll'
stdout 'The cursor is no longer valid'

# The event before the rejection and the event from the resumed query.
exec jq -r '[.hits.hits[]._source.event.id | select(. != null)] | sort | join(",")' got_docs.json
stdout '^00000a01-0000-4000-8000-000000000a01,00000a02-0000-4000-8000-000000000a02$'

# No pipeline errors.
exec jq '[.hits.hits[]._source | select(.event.kind == "pipeline_error")] | length' got_docs.json
stdout '^0$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down audit-mock
docker_down audit-oauth
