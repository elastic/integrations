# The Audit API rate-limits the results request (429 with Retry-After: 1). The
# CEL input's retry client (resource.retry.*) waits and repeats the request, so
# the poll recovers with no error document.
#
# Request flow (interval 5s, page_size 3, max_attempts 5):
#   poll 1: token -> createQuery -> cursor-ref-a
#           results(cursor-ref-a) -> 429, 429, then 2 events and cursor-ref-b
#                                    (the two retries are inside one poll)
#           results(cursor-ref-b) -> 0 events (caught up)
#
# The mock returns the responses of a rule in turn, so the first two calls for
# cursor-ref-a get 429 and the third gets the events. The mock log must show
# exactly two 429 responses, so the test cannot pass without the retry.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# The token URL is built from identity_id, so the OAuth mock uses the host name
# elastic-audit-test.id.cyberark.cloud over HTTPS. The policy trusts its
# self-signed certificate through the ssl var.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} audit-oauth
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} audit-mock

add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# 2 audit events and no more.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json
exec jq '.hits.total.value' got_docs.json
stdout '^2$'

# The retry recovered inside the poll: no error document.
exec jq '[.hits.hits[]._source | select(.error.message != null)] | length' got_docs.json
stdout '^0$'

exec jq -r '[.hits.hits[]._source.event.id | select(. != null)] | sort | join(",")' got_docs.json
stdout '^00000b01-0000-4000-8000-000000000b01,00000b02-0000-4000-8000-000000000b02$'

# No pipeline errors.
exec jq '[.hits.hits[]._source | select(.event.kind == "pipeline_error")] | length' got_docs.json
stdout '^0$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m

# docker_down prints the mock's request log. The API returned 429 exactly
# twice, and the program created only one query.
docker_down audit-mock
cp stdout mock.log
exec jq -n --rawfile log mock.log '$log | [scan("stream/results HTTP/[0-9.]+\" 429 ")] | length'
stdout '^2$'
exec jq -n --rawfile log mock.log '$log | [scan("stream/createQuery HTTP/[0-9.]+\" 200 ")] | length'
stdout '^1$'
docker_down audit-oauth
