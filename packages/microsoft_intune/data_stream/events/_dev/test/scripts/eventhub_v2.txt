# Verify that Microsoft Intune events read from a single Event Hub (AMQP,
# processor v2) are routed to the data stream for their log category.
#
# The fake EventHub server delivers eight test events:
#   - 2 AuditLogs events            → microsoft_intune.audit
#   - 2 Devices events              → microsoft_intune.managed_device
#   - 1 DeviceComplianceOrg event   → microsoft_intune.managed_device
#   - 1 OperationalLogs event       → microsoft_intune.events (not routed)
#   - 1 event that is not JSON      → microsoft_intune.events (pipeline error)
#   - 1 JSON event, not an object   → microsoft_intune.events (pipeline error)
#
# The server speaks AMQP 1.0 on port 5672 (EventHub emulator protocol) and
# Azure Blob Storage REST on port 10000 (checkpoint store). Its source lives
# in packages/microsoft_intune/_dev/scripts/azure-eh/.

[!external_stack] skip 'Skipping external stack test.'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Copy the fake EventHub server source into the working directory.
mkdir azure-eh
cp ${PACKAGE_ROOT}/_dev/scripts/azure-eh/docker-compose.yml azure-eh/docker-compose.yml
cp ${PACKAGE_ROOT}/_dev/scripts/azure-eh/go.mod azure-eh/go.mod
cp ${PACKAGE_ROOT}/_dev/scripts/azure-eh/main.go azure-eh/main.go

# Start the fake EventHub server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} azure-eh

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy pointing at the fake EventHub.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml EVENTS_DS

# Wait for all docs to arrive, then confirm the total holds.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 8 -confirm 5s -timeout 5m logs-microsoft_intune.audit-*,logs-microsoft_intune.managed_device-*,${EVENTS_DS}

# Each destination got its documents (docs already indexed, these return immediately).
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 0 -timeout 30s logs-microsoft_intune.audit-*
# Routed events have the tag of their data stream. The pipeline tests check
# that it replaces microsoft_intune-events; it is not checked here because the
# wildcard also matches documents left in other namespaces by earlier runs.
stdout '"microsoft_intune-audit"'
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 0 -timeout 30s logs-microsoft_intune.managed_device-*
stdout '"microsoft_intune-managed_device"'
# The device compliance event is parsed by the managed_device pipeline too.
stdout '"category":\s*"DeviceComplianceOrg"'
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 0 -timeout 30s ${EVENTS_DS}

# The unrouted event keeps the event.dataset set by the agent, its category
# and the microsoft_intune-events tag.
stdout '"event":\s*\{[^}]*"dataset":\s*"microsoft_intune\.events"'
stdout '"category":\s*"OperationalLogs"'
stdout '"microsoft_intune-events"'

# The two events that cannot be parsed stay in this data stream as pipeline
# errors. The raw event is kept in message and event.original although
# Preserve original event is disabled, and the temporary field is removed.
stdout '"kind":\s*"pipeline_error"'
stdout 'json_message_to_tmp_json_3430dc7f'
stdout '"message":\s*"not a json message"'
stdout '"original":\s*"not a json message"'
stdout '"message":\s*"12345"'
stdout '"original":\s*"12345"'
stdout '"preserve_original_event"'
! stdout '"_tmp"'

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${EVENTS_DS}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down azure-eh
