# Verify that microsoft_intune managed_device events delivered over Event Hub
# (AMQP, processor v2) flow through the ingest pipeline and land in the
# correct data stream.
#
# The fake EventHub server delivers two test events:
#   - a Windows corporate device (Lenovo, compliant)
#   - a Windows corporate device (VMware, compliant)
#
# The server speaks AMQP 1.0 on port 5672 (EventHub emulator protocol) and
# Azure Blob Storage REST on port 10000 (checkpoint store). Its source lives
# in packages/microsoft_intune/_dev/scripts/azure-eh-managed-device/.

[!external_stack] skip 'Skipping external stack test.'

# Connect to the running stack.
use_stack -profile ${CONFIG_PROFILES}/${PROFILE}

# Install an agent.
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME

# Copy the fake EventHub server source into the working directory.
mkdir azure-eh
cp ${PACKAGE_ROOT}/_dev/scripts/azure-eh-managed-device/docker-compose.yml azure-eh/docker-compose.yml
cp ${PACKAGE_ROOT}/_dev/scripts/azure-eh-managed-device/go.mod azure-eh/go.mod
cp ${PACKAGE_ROOT}/_dev/scripts/azure-eh-managed-device/main.go azure-eh/main.go

# Start the fake EventHub server.
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} azure-eh

# Add the package resources.
add_package -profile ${CONFIG_PROFILES}/${PROFILE}

# Add the data stream policy pointing at the fake EventHub.
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DEVICE_DS

# Wait for the test events to arrive.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 2 -confirm 5s -timeout 5m logs-microsoft_intune.managed_device-*

# Clean up.
remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DEVICE_DS}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down azure-eh
