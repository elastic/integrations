# Test that an HTTP 200 response with a non-JSON body part way through a
# paginated window is reported as an error, drops the page token, and that the
# next poll restarts the window from its first page.
#
# The mock returns a first page with a next_page_token. A request carrying that
# token gets a 200 response with an XML body. Before the fix the stale token
# stayed in the input state and every poll repeated the failing request, so the
# second page of first-page data below was never fetched. With the fix the
# following poll requests the first page again and receives the "recovered"
# record.

[!external_stack] skip 'Skipping external stack test.'
[!exec:jq] skip 'Skipping test requiring absent jq command'

use_stack -profile ${CONFIG_PROFILES}/${PROFILE}
install_agent -profile ${CONFIG_PROFILES}/${PROFILE} -network_name NETWORK_NAME
docker_up -profile ${CONFIG_PROFILES}/${PROFILE} -network ${NETWORK_NAME} zoom-mock
add_package -profile ${CONFIG_PROFILES}/${PROFILE}
add_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} test_config.yaml DATA_STREAM_NAME

# Expect the first page record, the error event for the non-JSON response and
# the record returned when the window is restarted.
get_docs -profile ${CONFIG_PROFILES}/${PROFILE} -want 3 -confirm 15s -timeout 5m ${DATA_STREAM_NAME}
cp stdout got_docs.json

exec jq -r '[.hits.hits[]._source.user.name | select(. != null)] | sort | join(",")' got_docs.json
stdout '^Alice Adams,Jill Chill$'

exec jq -r '[.hits.hits[]._source.error.message | select(. != null)] | length' got_docs.json
stdout '^1$'

remove_package_policy -profile ${CONFIG_PROFILES}/${PROFILE} ${DATA_STREAM_NAME}
uninstall_agent -profile ${CONFIG_PROFILES}/${PROFILE} -timeout 1m
docker_down zoom-mock

-- test_config.yaml --
input: cel
vars:
  url: http://zoom-mock:8090
  token_url: http://zoom-mock:8090
  account_id: test-account-id
  client_id: test-client-id
  client_secret: test-client-secret
data_stream:
  vars:
    interval: 5s
    initial_interval: 24h
    activity_type: All Activities
    page_size: 300
    tags:
      - forwarded
      - zoom-meeting_activity
